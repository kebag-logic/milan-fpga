def command(argv: list[str], log: Path, cwd: Path, allow_fail: bool = False) -> tuple[int, str]:
    started = dt.datetime.now(dt.timezone.utc).isoformat()
    p = subprocess.run(["rtk", "proxy", *argv], cwd=cwd, text=True,
                       capture_output=True,
                       env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1",
                            "MAKEFLAGS": "-j2"})
    ended = dt.datetime.now(dt.timezone.utc).isoformat()
    log.parent.mkdir(parents=True, exist_ok=True)
    log.write_text((p.stdout + p.stderr).replace(str(cwd), "<CHECKOUT>"))
    COMMANDS.append(dict(argv=scrub(argv, cwd), start_utc=started,
                         end_utc=ended, exit=p.returncode,
                         log=str(log.relative_to(HERE))))
    (HERE / "commands.json").write_text(json.dumps(COMMANDS, indent=2) + "\n")
    print(f"  {log.name}: exit={p.returncode}", flush=True)
    if p.returncode and not allow_fail:
        raise RuntimeError(f"command failed, see {log}")
    return p.returncode, p.stdout + p.stderr


def once(text: str, old: str, new: str, what: str) -> str:

    return text.replace(old, new, 1)

SCHEDULER = r'''
/* ---- issue #419 external scheduler (temporary, diagnostic only) --------
 * `be` remains for the deadline counters and the boot walk; the durability,
 * validity, bracket and capture state the firmware reads come from the RTL. */
#define DIAG_MAX_ACTS 8
static struct { char trig[24]; char act[16]; int fired; } diag_acts[DIAG_MAX_ACTS];
static int diag_nact;
static uint64_t diag_arm_ns;   /* when the capture window opened */

static void sync_rtl(void)
{
    uint32_t s = rtl_status();
    be.backed = (s >> 6) & 1u;
    be.img_valid = (s >> 7) & 1u;
    be.dirty = (s >> 8) & 1u;
    be.stale = (s >> 9) & 1u;
    be.commit_busy = (s >> 10) & 1u;
    be.verdict = (s >> 12) & 15u;
}

static void diag_add(const char *spec)
{
    const char *colon = strchr(spec, ':');
    if (!colon || diag_nact == DIAG_MAX_ACTS) {
        fprintf(stderr, "DIAG: --at wants TRIGGER:ACTION\n");
        exit(2);
    }
    if ((size_t)(colon - spec) >= sizeof diag_acts[0].trig ||
        strlen(colon + 1) >= sizeof diag_acts[0].act) {
        fprintf(stderr, "DIAG: --at name too long\n");
        exit(2);
    }
    memcpy(diag_acts[diag_nact].trig, spec, (size_t)(colon - spec));
    diag_acts[diag_nact].trig[colon - spec] = '\0';
    strcpy(diag_acts[diag_nact].act, colon + 1);
    diag_nact++;
}

static int diag_pending(const char *trigger)
{
    int i;
    for (i = 0; i < diag_nact; ++i)
        if (!diag_acts[i].fired && strcmp(diag_acts[i].trig, trigger) == 0)
            return 1;
    return 0;
}

/* The full-image witness: before an injection at the post-verify hook, the
 * acknowledged flash bytes and the live image must still be equal. */
static void diag_witness(void)
{
    const uint8_t *flash = nvm_host_flash + NVM_HOST_JOURNAL_OFFSET;
    const uint8_t *flash_b = flash + 0x10000u;
    const uint8_t *ddr = nvm_host_ddr + NVM_HOST_IMAGE_OFF;
    int match = memcmp(flash, ddr, DIAG_IMAGE_LEN) == 0 ||
                memcmp(flash_b, ddr, DIAG_IMAGE_LEN) == 0;
    printf("DIAG post_verify_sequence_store host_ns=%llu rtl_cycle=%llu full_image_match=%d\n",
        (unsigned long long)now_ns, (unsigned long long)rtl_cycles(), match);
    if (!match) { fprintf(stderr, "DIAG: verification witness differs\n"); exit(2); }
}

static void diag_do(const char *act, int coincident)
{
    unsigned k = (act[strlen(act) - 1] == '1') ? 1u : 0u;
    printf("DIAG action=%s record_id=0x%02x host_ns=%llu rtl_cycle=%llu coincident=%d\n",
        act, DIAG_RID(k), (unsigned long long)now_ns,
        (unsigned long long)rtl_cycles(), coincident);
    if (!strncmp(act, "write", 5))
        rtl_record_update(DIAG_OFF(k), DIAG_FRAME(k), DIAG_FRAME_LEN, coincident);
    else if (!strncmp(act, "erase", 5))
        rtl_record_erase(DIAG_RID(k));
    else if (!strncmp(act, "close", 5))
        rtl_record_program(DIAG_OFF(k), DIAG_FRAME(k), DIAG_FRAME_LEN, coincident);
    else if (!strncmp(act, "fail", 4)) {
        rtl_record_erase(DIAG_RID(k));
        rtl_record_program_fail(DIAG_RID(k), DIAG_FRAME(k), DIAG_FRAME_LEN);
    } else if (!strcmp(act, "dupack")) {
        rtl_csr_write(4, 2u);
    } else {
        fprintf(stderr, "DIAG: unknown action %s\n", act);
        exit(2);
    }
    sync_rtl();
}

static void diag_fire(const char *trigger)
{
    int i;
    for (i = 0; i < diag_nact; ++i) {
        if (diag_acts[i].fired || strcmp(diag_acts[i].trig, trigger) != 0)
            continue;
        diag_acts[i].fired = 1;
        printf("DIAG trigger=%s\n", trigger);
        diag_do(diag_acts[i].act, strcmp(trigger, "on-ack") == 0);
    }
}
'''

        fences[tag] = n
        assert n == 2, (tag, n)
        src = work / f"milan_baremetal.{tag}.host.c"
        src.write_text(adapted)
        for pe in ([False, True] if tag == "proto" else [False]):
            key = tag + ("+preerase" if pe else "")
            obj = work / f"fw.{key.replace('+', '_')}.o"
            command(common + ["-Wno-format"]
                    + (["-DMILAN_NVM_PREERASE"] if pe else [])
                    + ["-c", str(src), "-o", str(obj)],
                    logs / f"compile-fw-{key.replace('+', '_')}.log", root)
            fw_objects[key] = obj

    # ---- the unit bench: the tracked module, the prototype, six mutants ---
    unit_results = {}
    unit_variants = [("pristine", root / RTL), ("proto", HERE / "prototype/KL_nvm_backend.proto.sv")]
    unit_variants += [(p.stem, p) for p in sorted((HERE / "mutations").glob("*.sv"))]
    print("unit bench")
    for tag, src in unit_variants:
        mdir = HERE / f"unit/obj-{tag}"
        shutil.rmtree(mdir, ignore_errors=True)
        command(["verilator", "--cc", "--exe", "--build", "-j", "2",
                 "--top-module", "KL_nvm_backend", "--Mdir", str(mdir),
                 "-Wall", "-Wno-fatal", "-GCLK_HZ_P=50000000", *SHAPE_PARAMS,
                 "-CFLAGS", "-std=c++17 -O1", str(src),
                 str(HERE / "unit/unit_main.cpp"), "-o", "unit"],
                logs / f"unit-build-{tag}.log", root)
        variant = "proto" if tag != "pristine" else "pristine"
        rc, out = command([str(mdir / "unit"), "--variant", variant],
                          logs / f"unit-{tag}.log", root, allow_fail=True)
        failed = re.findall(r"^UNIT (\S+)\s+FAIL", out, re.M)
        passed = re.findall(r"^UNIT (\S+)\s+PASS", out, re.M)
        unit_results[tag] = dict(exit=rc, failed=failed, n_pass=len(passed))
        print(f"    {tag}: {len(passed)} pass, {len(failed)} fail")

    # ---- the co-simulation builds ----------------------------------------
    builds = {
        "baseline": (root / RTL, "pristine"),
        "prototype": (HERE / "prototype/KL_nvm_backend.proto.sv", "proto"),
        "prototype+preerase": (HERE / "prototype/KL_nvm_backend.proto.sv", "proto+preerase"),
        "mismatch": (HERE / "prototype/KL_nvm_backend.proto.sv", "pristine"),
        "mutant-M1": (HERE / "mutations/M1_window_never_voids.sv", "proto"),
    }
    if args.only:
        builds = {k: v for k, v in builds.items() if k == args.only}
    results: dict[str, dict] = {}
    for build, (rtl_src, fw_key) in builds.items():
        slug = build.replace("+", "_")
        mdir = work / f"obj-{slug}"
        shutil.rmtree(mdir, ignore_errors=True)
        print(f"co-simulation build {build}")
        command(["verilator", "--cc", "--exe", "--build", "-j", "2",
                 "--top-module", "KL_nvm_backend", "--Mdir", str(mdir),
                 "-Wall", "-Wno-fatal", "-GCLK_HZ_P=50000000", *SHAPE_PARAMS,
                 "-CFLAGS", f"-std=c++17 -O1 -I{ref.STUBS} -I{work} {defs[-1]}",
                 "-LDFLAGS", f"{work / 'host.o'} {fw_objects[fw_key]}",
                 str(rtl_src), str(HERE / "harness/proto_bridge.cpp"),
                 "-o", "cosim"], logs / f"cosim-build-{slug}.log", root)
        results[build] = {}
        for case in CASES:
            if fw_key.startswith("pristine") and any(
                    t in " ".join(CASES[case].get("at", []))
                    for t in PROTO_ONLY_TRIGGERS):
                results[build][case] = dict(skipped="trigger not emitted by the "
                                            "tracked firmware")
                continue
            out = HERE / f"cases/{slug}/{case}"
            out.mkdir(parents=True, exist_ok=True)
            rc, text = command([str(mdir / "cosim")]
                               + case_argv(case, out, preload),
                               logs / f"{slug}-{case}.log", root, allow_fail=True)
            results[build][case] = grade(ref, donor, ident, expect, case, out,
                                         img_len, off, old, new,
                                         erased, rids, text, rc)
        n_bad = sum(1 for c in results[build].values() if c.get("violations"))
        print(f"    {build}: {n_bad} case(s) with contract violations")

    after = {p: sha((root / p).read_bytes()) for p in PRISTINE}
    assert before == after
    (HERE / "results.json").write_text(json.dumps(dict(
        root_head=head, donor_pin=pin, rtl_clock_hz=50_000_000,
        host_model_clock_hz=100_000_000, image_bytes=img_len,
        record_offsets={hex(rids[k]): off[k] for k in (0, 1)},
        fence_adaptations=fences, production_sha256_before=before,
        production_sha256_after=after, unit=unit_results, cosim=results),
        indent=2) + "\n")
    all_hashes = {str(p.relative_to(HERE)): sha(p.read_bytes())
                  for p in HERE.rglob('*')
                  if p.is_file() and 'obj' not in p.parts and 'build' not in p.parts
                  and p.name not in ('hashes.json', 'commands.json')}
    (HERE / "hashes.json").write_text(json.dumps(all_hashes, indent=2) + "\n")
    print("\nWROTE results.json")


def grade(ref, donor, ident, expect, case: str, out: Path, img_len: int,
          off: list[int], old: list[bytes], new: list[bytes],
          erased: dict[int, bytes], rids: list[int], text: str, rc: int) -> dict:
    """The independent oracle: decode both slots with the repository's own
    decoder and read the record bytes out of the image, never out of a status
    bit. `violations` names contract statements the run broke."""
    want_record, want_dirty = CASES[case]["want"], CASES[case]["dirty"]
    report: dict = dict(exit=rc, violations=[], checkpoints={})
    frames = {"old": old[0], "new": new[0], "erased": erased[rids[0]]}
    for phase in ("first", "final"):
        slots = {}
        for letter in ("a", "b"):
            blob = (out / f"slot-{letter}-{phase}.bin").read_bytes()[:img_len]
            vd = ref.klj2_decode(blob, donor, ident, expect)[0]
            seq = int.from_bytes(blob[8:12], "little") if vd == ref.VD_OK else -1
            name = next((k for k, v in frames.items()
                         if blob[off[0]:off[0] + len(v)] == v), "other")
            slots[letter] = dict(verdict=int(vd), seq=seq, record0=name,
                                 sha256=sha(blob))
        newest = max(("a", "b"), key=lambda s: slots[s]["seq"])
        report["checkpoints"][phase] = dict(
            slots=slots, newest=newest,
            any_ok=any(s["seq"] >= 0 for s in slots.values()))
        # A torn target slot is legal at any instant; losing the last complete
        # image is not (design page section 7).
        if phase == "final" and report["checkpoints"]["first"]["any_ok"] \
                and not report["checkpoints"]["final"]["any_ok"]:
            report["violations"].append(
                "final: no slot decodes a complete image although one did at "
                "the first checkpoint")
        if slots[newest]["seq"] >= 0 and slots[newest]["record0"] == "erased":
            report["violations"].append(
                f"{phase}: the newest accepted slot holds an ERASED record 0x20 "
                f"although the record has been written")
    summaries = [dict((k, int(v)) for k, v in
                      (x.split("=") for x in fields.split()))
                 for fields in re.findall(r"^HOST (hb=.*)$", text, re.M)]
    if not summaries:
        report["violations"].append("no HOST summary line: the run did not finish")
        return report
    report["host_first"], report["host_final"] = summaries[0], summaries[-1]
    final = report["checkpoints"]["final"]
    newest = final["slots"][final["newest"]]
    if not final["any_ok"]:
        report["violations"].append(
            "final: neither slot decodes a complete image")
    elif newest["record0"] != want_record:
        report["violations"].append(
            f"final: the newest accepted slot holds the {newest['record0']} "
            f"record 0x20, expected {want_record}")
    if report["host_final"]["dirty"] != want_dirty:
        report["violations"].append(
            f"final: dirty={report['host_final']['dirty']}, expected {want_dirty}")
    if report["host_final"]["losses"] != 0:
        report["violations"].append(
            f"the host model recorded {report['host_final']['losses']} writer "
            f"loss(es); no deadline should have lapsed")
    if CASES[case].get("slow") and report["host_final"]["max_hb_gap_ms"] > 500:
        report["violations"].append(
            f"the heartbeat gap reached {report['host_final']['max_hb_gap_ms']} "
            f"ms through the 3 s erase, above the 500 ms the contract requires")
    report["acks"] = report["host_final"]["acks"]
    report["erases"] = report["host_final"]["erases"]
    report["max_hb_gap_ms"] = report["host_final"]["max_hb_gap_ms"]
    report["ack_refused_lines"] = len(re.findall(r"NOT acknowledged", text))
    report["gap_held_lines"] = len(re.findall(r"has been blank for", text))
    report["deferred_lines"] = len(re.findall(r"deferred", text))
    report["capture_windows_us"] = [int(x) for x in
                                    re.findall(r"DIAG window_us=(\d+)", text)]
    return report
