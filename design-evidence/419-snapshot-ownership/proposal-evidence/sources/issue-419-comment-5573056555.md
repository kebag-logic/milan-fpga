// Issue #419 design prototype: an RTL-only edge bench for the acknowledgement
// contract. It drives KL_nvm_backend's real device, memory and control faces
// and nothing else; no firmware, no flash and no host model take part, so the
// numbers it prints are backend clock edges and not milliseconds of anything.
//
// The same source grades the tracked module and the prototype: --variant
// selects which contract the expectations come from, so the baseline column
// in the report is measured rather than quoted.
#include "VKL_nvm_backend.h"
#include "verilated.h"

#include <cstdint>
#include <cstdio>
#include <cstring>
#include <memory>
#include <string>
#include <vector>

namespace {

constexpr uint32_t kBase = 0x40000000;
constexpr uint32_t kAreaLen = 2652;   // the 1x1 shape's record area, bytes
constexpr uint32_t kBindOff = 122;    // record 0x20 inside the record area
constexpr unsigned kBindLen = 28;     // its framed length

std::unique_ptr<VKL_nvm_backend> dut;
std::vector<uint8_t> mem(0x10000, 0xA5);
uint64_t cycles = 0;
int failures = 0;
bool proto = false;

struct Write {
  bool busy = false;
  int delay = 0;
  uint32_t addr = 0;
  uint64_t data = 0;
  uint8_t strb = 0;
} wr;

bool wr_fail_next = false;   // make the NEXT accepted memory write report err

void edge() {
  dut->clk_i = 0;
  dut->mem_wr_ready_i = !wr.busy;
  dut->mem_wr_done_i = wr.busy && wr.delay == 0;
  dut->mem_wr_err_i = dut->mem_wr_done_i && wr_fail_next;
  dut->eval();
  const bool take = dut->mem_wr_valid_o && dut->mem_wr_ready_i;
  const bool finish = dut->mem_wr_done_i;
  if (finish) {
    if (!wr_fail_next) {
      for (unsigned n = 0; n < 8; ++n)
        if (wr.strb & (1u << n))
          mem[(wr.addr - kBase + n) & 0xFFFF] = uint8_t(wr.data >> (56 - 8 * n));
    } else {
      wr_fail_next = false;
    }
    wr.busy = false;
  } else if (wr.busy) {
    --wr.delay;
  }
  if (take)
    wr = {true, 2, dut->mem_wr_addr_o, dut->mem_wr_data_o,
          uint8_t(dut->mem_wr_strb_o)};
  dut->clk_i = 1;
  dut->eval();
  ++cycles;
  dut->clk_i = 0;
  dut->eval();
}

uint32_t status() {
  const auto keep_sel = dut->csr_sel_i;
  const auto keep_we = dut->csr_we_i;
  const auto keep_addr = dut->csr_addr_i;
  dut->csr_sel_i = 1;
  dut->csr_we_i = 0;
  dut->csr_addr_i = 3;
  dut->eval();
  const uint32_t v = dut->csr_rdata_o;
  dut->csr_sel_i = keep_sel;
  dut->csr_we_i = keep_we;
  dut->csr_addr_i = keep_addr;
  dut->eval();
  return v;
}

struct Bits {
  unsigned dev_busy, cfg, backed, valid, dirty, stale, bracket;
  unsigned armed, cap_valid, gap, ack_ref, tag;
};

Bits bits() {
  const uint32_t s = status();
  return Bits{(s >> 4) & 1u,  (s >> 5) & 1u,  (s >> 6) & 1u,  (s >> 7) & 1u,
              (s >> 8) & 1u,  (s >> 9) & 1u,  (s >> 10) & 1u, (s >> 16) & 1u,
              (s >> 17) & 1u, (s >> 18) & 1u, (s >> 19) & 1u, (s >> 24) & 0xffu};
}

void csr(unsigned addr, uint32_t value) {
  dut->csr_sel_i = 1;
  dut->csr_we_i = 1;
  dut->csr_addr_i = addr;
  dut->csr_wdata_i = value;
  edge();
  dut->csr_sel_i = 0;
  dut->csr_we_i = 0;
}

// One strobe of the control word: [0] heartbeat, [1] acknowledge, [2] start,
// [3] arm (the prototype's).
void strobe(uint32_t v) { csr(4, v); }

void reset() {
  dut->rst_n = 0;
  dut->change_i = 0;
  dut->dev_req_i = dut->dev_wvalid_i = dut->dev_rready_i = 0;
  dut->mem_req_ready_i = dut->mem_rsp_valid_i = 0;
  dut->mem_rsp_last_i = 1;
  dut->mem_rsp_err_i = 0;
  dut->csr_sel_i = dut->csr_we_i = 0;
  for (int n = 0; n < 4; ++n) edge();
  dut->rst_n = 1;
  edge();
}

void configure() {
  csr(0, kBase);
  csr(1, kAreaLen);
  csr(3, 0x10u);              // firmware says: validated, VD_OK
}

// Run one device command. `ack_gap` >= 0 presents the acknowledgement strobe
// that many edges after the completion pulse; 0 is the same edge.
// Returns the edge on which dev_done_o was observed.
uint64_t command(unsigned op, unsigned rid, const std::vector<uint8_t> &frame,
                 int ack_gap = -1, bool fail_first_write = false) {
  dut->dev_req_i = 1;
  dut->dev_op_i = op;
  dut->dev_region_i = rid;
  dut->dev_offset_i = 0;
  dut->dev_len_i = (op == 2) ? 0 : uint16_t(frame.size());
  dut->dev_wvalid_i = 0;
  edge();
  if (!dut->dev_gnt_o) {
    std::printf("UNIT FAIL command rid=0x%02x op=%u: no grant\n", rid, op);
    ++failures;
    return 0;
  }
  dut->dev_req_i = 0;
  size_t sent = 0;
  bool armed_fail = fail_first_write;
  uint64_t done_edge = 0;
  for (unsigned guard = 0; guard < 20000; ++guard) {
    dut->dev_wvalid_i = (op == 1) && sent < frame.size();
    if (dut->dev_wvalid_i) dut->dev_wdata_i = frame[sent];
    dut->eval();
    if (armed_fail && dut->mem_wr_valid_o) {
      wr_fail_next = true;
      armed_fail = false;
    }
    const bool took = dut->dev_wvalid_i && dut->dev_wready_o;
    const bool done = dut->dev_done_o;
    const bool err = dut->dev_err_o;
    if (done && ack_gap == 0) strobe(2);   // acknowledge on the exact edge
    edge();
    if (took) ++sent;
    if (done || err) {
      dut->dev_wvalid_i = 0;
      done_edge = cycles;
      for (int k = 0; k < ack_gap; ++k) edge();
      if (ack_gap > 0) strobe(2);
      return done_edge;
    }
  }
  std::printf("UNIT FAIL command rid=0x%02x op=%u: never terminated\n", rid, op);
  ++failures;
  return 0;
}

std::vector<uint8_t> frame(uint8_t flags) {
  std::vector<uint8_t> f(kBindLen, 0);
  f[0] = 0x17; f[1] = 0x22; f[2] = 0x02; f[3] = 0x20;
  f[4] = 0x00; f[5] = 0x14; f[6] = 0x00; f[7] = 0x00;
  f[8] = flags;
  return f;
}

void check(const char *name, bool ok, const char *detail) {
  std::printf("UNIT %-28s %s  %s\n", name, ok ? "PASS" : "FAIL", detail);
  if (!ok) ++failures;
}

void u1_reset_and_bare_ack() {
  reset();
  configure();
  Bits b = bits();
  char d[256];
  std::snprintf(d, sizeof d, "after reset armed=%u cap_valid=%u gap=%u dirty=%u tag=0x%02x",
                b.armed, b.cap_valid, b.gap, b.dirty, b.tag);
  check("U1.reset_state", b.armed == 0 && b.cap_valid == 0 && b.gap == 0 &&
        b.dirty == 0, d);
  command(2, 0x20, {});
  command(1, 0x20, frame(0x03));          // a completed record change
  strobe(2);                              // acknowledge with no window at all
  b = bits();
  std::snprintf(d, sizeof d, "no window opened, ack issued: dirty=%u ack_refused=%u",
                b.dirty, b.ack_ref);
  check("U1.ack_without_window", b.dirty == 1 && b.ack_ref == 1, d);
}

// The measured shape of the defect: how many edges of separation between a
// completed record WRITE and the acknowledgement still lose the change.
void u2_ack_gap_sweep() {
  const int gaps[] = {0, 1, 2, 4, 8, 64};
  for (int g : gaps) {
    reset();
    configure();
    command(2, 0x20, {});                        // ERASE the whole record
    command(1, 0x20, frame(0x03));               // WRITE the old frame
    strobe(8);                                   // warm-up window (ignored today)
    strobe(2);                                   // retire it: both reach dirty=0
    // open a window, then let a record update complete inside it
    strobe(8);                                   // ARM (a no-op on the tracked module)
    Bits before = bits();
    command(2, 0x20, {});
    command(1, 0x20, frame(0x01), g);            // WRITE, then ACK g edges later
    Bits after = bits();
    char d[256];
    std::snprintf(d, sizeof d,
        "gap=%2d edges: armed_before=%u cap_valid_before=%u -> dirty=%u ack_refused=%u",
        g, before.armed, before.cap_valid, after.dirty, after.ack_ref);
    // CONTRACT: a change that completes inside the window is never retired,
    // whatever its separation from the acknowledgement, and an
    // acknowledgement that retires nothing says so.
    check((std::string("U2.ack_gap_") + std::to_string(g)).c_str(),
          after.dirty == 1 && after.ack_ref == 1, d);
  }
}

void u3_clean_window_retires() {

}

void u5_erase_gap() {
  reset();
  configure();
  command(2, 0x20, {});
  command(1, 0x20, frame(0x03));
  strobe(8);
  strobe(2);                                     // steady state: nothing pending
  Bits b0 = bits();
  command(2, 0x20, {});                          // the producer blanks the span
  Bits b1 = bits();
  strobe(8);                                     // try to capture over the blank
  Bits b2 = bits();
  strobe(2);
  Bits b3 = bits();
  command(1, 0x20, frame(0x01));                 // the producer closes the span
  Bits b4 = bits();
  strobe(8);
  strobe(2);                                     // and now it converges
  Bits b5 = bits();
  char d[320];
  std::snprintf(d, sizeof d,
      "steady dirty=%u; after ERASE gap=%u dirty=%u; arm cap_valid=%u; ack dirty=%u; after WRITE gap=%u; final dirty=%u",
      b0.dirty, b1.gap, b1.dirty, b2.cap_valid, b3.dirty, b4.gap, b5.dirty);
  // CONTRACT: a span the producer has blanked is not a saveable state, and
  // an acknowledgement taken over one retires nothing; once the producer
  // closes it, an ordinary window converges.
  check("U5.blank_span_not_retired",
        b0.dirty == 0 && b1.dirty == 1 && b3.dirty == 1 && b3.ack_ref == 1 &&
        b5.dirty == 0 && b5.ack_ref == 0, d);
}

void u6_arm_while_busy() {
  reset();
  configure();
  command(2, 0x20, {});
  command(1, 0x20, frame(0x03));
  strobe(8);
  strobe(2);                                     // steady state
  // arm while an operation is already in flight
  dut->dev_req_i = 1;
  dut->dev_op_i = 2;
  dut->dev_region_i = 0x20;
  dut->dev_offset_i = 0;
  dut->dev_len_i = 0;
  edge();
  dut->dev_req_i = 0;
  edge();
  Bits busy = bits();
  strobe(8);                                     // the arm lands mid-operation
  Bits armed = bits();
  for (unsigned g = 0; g < 20000 && !dut->dev_done_o; ++g) edge();
  edge();
  command(1, 0x20, frame(0x01));                 // the update completes
  strobe(2);
  Bits after = bits();
  char d[256];
  std::snprintf(d, sizeof d,
      "dev_busy=%u at the arm, cap_valid=%u -> after the update dirty=%u",
      busy.dev_busy, armed.cap_valid, after.dirty);
  // CONTRACT: a window opened over an operation in flight cannot retire what
  // that operation completes.
  check("U6.arm_while_busy",
        busy.dev_busy == 1 && after.dirty == 1 && after.ack_ref == 1, d);
}

void u7_gap_stuck() {

  strobe(2);

  reset();
  configure();
  Bits b4 = bits();

// The exact-edge case in isolation: a WRITE with no preceding ERASE, so the

// SPDX-License-Identifier: CERN-OHL-W-2.0
// Issue #419 design prototype: a temporary external device/DRAM scheduler
// around the backend RTL. Derived from the frozen issue #418 diagnostic
// bridge (proposal-evidence/harness/rtl_bridge.cpp.418) and extended with
// the halves of a producer logical operation, a failing memory write and a
// reset pulse, so the ERASE-to-WRITE gap and the failure paths can be
// scheduled. It drives only public module ports.
#include "VKL_nvm_backend.h"
#include "verilated.h"
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <memory>
#include <vector>
extern "C" {
#include "nvm_host.h"
}

namespace {
constexpr uint32_t kBase = 0x40000000;
constexpr uint64_t kClockNs = 20; // 50 MHz backend clock, separate from host time.
std::unique_ptr<VKL_nvm_backend> dut;
uint64_t cycles = 0;
uint64_t write_done_cycle = 0;
uint64_t ack_cycle = 0;
unsigned accepted = 0;
unsigned bytes_committed = 0;
bool fail_next_mem_write = false;
struct Write {
  bool busy = false;
  int delay = 0;
  uint32_t addr = 0;
  uint64_t data = 0;
  uint8_t strb = 0;
  uint64_t accept_cycle = 0;
  bool poison = false;
} wr;

[[noreturn]] void fail(const char *why) {
  std::fprintf(stderr, "RTL PROTOTYPE ERROR: %s\n", why);
  std::exit(2);
}

uint8_t *area() { return nvm_host_ddr + NVM_HOST_IMAGE_OFF + 40; }

void edge() {
  dut->clk_i = 0;
  dut->mem_wr_ready_i = !wr.busy;
  dut->mem_wr_done_i = wr.busy && wr.delay == 0;
  dut->mem_wr_err_i = dut->mem_wr_done_i && wr.poison;
  dut->eval();
  const bool take = dut->mem_wr_valid_o && dut->mem_wr_ready_i;
  const bool finish = dut->mem_wr_done_i;
  if (dut->dev_done_o && dut->dev_op_i == 1) write_done_cycle = cycles + 1;
  if (dut->csr_sel_i && dut->csr_we_i && dut->csr_addr_i == 4 &&
      (dut->csr_wdata_i & 2)) ack_cycle = cycles + 1;
  if (finish) {
    std::printf("BRIDGE accept_cycle=%llu commit_cycle=%llu addr=0x%08x strb=0x%02x data=0x%016llx err=%d\n",
        (unsigned long long)wr.accept_cycle, (unsigned long long)(cycles + 1),
        wr.addr, wr.strb, (unsigned long long)wr.data, wr.poison ? 1 : 0);
    if (!wr.poison) {
      for (unsigned n = 0; n < 8; ++n) {
        if (!(wr.strb & (1u << n))) continue;
        if (wr.addr < kBase || wr.addr - kBase + n >= 0x10000 - 44)
          fail("memory write outside staged record area");
        area()[wr.addr - kBase + n] = uint8_t(wr.data >> (56 - 8 * n));
        ++bytes_committed;
      }
    }
    wr.busy = false;
  } else if (wr.busy) {
    --wr.delay;
  }
  if (take) {
    if (wr.busy) fail("bridge accepted overlapping write");
    wr = {true, 2, dut->mem_wr_addr_o, dut->mem_wr_data_o,
          uint8_t(dut->mem_wr_strb_o), cycles + 1, fail_next_mem_write};
    fail_next_mem_write = false;
  }
  dut->clk_i = 1;
  dut->eval();
  ++cycles;
  dut->clk_i = 0;
  dut->eval();
}

void drive_csr(unsigned addr, uint32_t value) {
  dut->csr_sel_i = dut->csr_we_i = 1;
  dut->csr_addr_i = addr;
  dut->csr_wdata_i = value;
}
void clear_csr() { dut->csr_sel_i = dut->csr_we_i = 0; }

uint32_t status() {
  const auto old = dut->csr_addr_i;
  const auto sel = dut->csr_sel_i;
  const auto we = dut->csr_we_i;
  dut->csr_sel_i = 1;
  dut->csr_we_i = 0;
  dut->csr_addr_i = 3;
  dut->eval();
  const uint32_t result = dut->csr_rdata_o;
  dut->csr_addr_i = old;
  dut->csr_sel_i = sel;
  dut->csr_we_i = we;
  dut->eval();
  return result;
}

void log(const char *tag) {
  const uint32_t s = status();
  std::printf("RTL tag=%s cycle=%llu rtl_ns=%llu status=0x%08x backed=%u dirty=%u stale=%u "
              "bracket=%u dev_busy=%u armed=%u cap_valid=%u gap=%u ack_refused=%u tag_byte=0x%02x change_pin=%u\n",
      tag, (unsigned long long)cycles, (unsigned long long)(cycles * kClockNs), s,
      (s >> 6) & 1, (s >> 8) & 1, (s >> 9) & 1, (s >> 10) & 1, (s >> 4) & 1,
      (s >> 16) & 1, (s >> 17) & 1, (s >> 18) & 1, (s >> 19) & 1,
      (s >> 24) & 0xff, unsigned(dut->change_i));
}

// One device command on the real device face. Returns when it terminates.
void command(unsigned op, unsigned rid, const std::vector<uint8_t> &frame,
             bool coincident_ack, bool expect_err) {
  dut->dev_req_i = 1;
  dut->dev_op_i = op;
  dut->dev_region_i = rid;
  dut->dev_offset_i = 0;
  dut->dev_len_i = op == 2 ? 0 : frame.size();
  dut->dev_wvalid_i = 0;
  edge();
  if (!dut->dev_gnt_o) fail("backend did not grant idle device request");
  std::printf("DEVICE grant_cycle=%llu op=%u record_id=0x%02x offset=0 len=%u bracket=%u gap=%u\n",
      (unsigned long long)cycles, op, rid, unsigned(dut->dev_len_i),
      (status() >> 10) & 1, (status() >> 18) & 1);
  ++accepted;
  dut->dev_req_i = 0;
  size_t sent = 0;
  for (unsigned guard = 0; guard < 100000; ++guard) {
    dut->dev_wvalid_i = op == 1 && sent < frame.size();
    if (dut->dev_wvalid_i) dut->dev_wdata_i = frame[sent];
    dut->eval();
    const bool took = dut->dev_wvalid_i && dut->dev_wready_o;
    const bool done = dut->dev_done_o;
    const bool err = dut->dev_err_o;
    if (done && coincident_ack) {
      if (op != 1) fail("ACK incorrectly scheduled on erase completion");
      drive_csr(4, 2);
      log("write_done_and_ack_preedge");
    }
    edge();
    if (took) ++sent;
    if (done || err) {
      dut->dev_wvalid_i = 0;
      clear_csr();
      if (err && !expect_err) fail("device operation reported err");
      if (done && expect_err) fail("expected the device operation to fail");
      if (coincident_ack && write_done_cycle != ack_cycle)
        fail("coincident stimulus missed exact edge");
      std::printf("DEVICE terminal_cycle=%llu op=%u record_id=0x%02x bytes_sent=%zu done=%d err=%d\n",
          (unsigned long long)cycles, op, rid, sent, done ? 1 : 0, err ? 1 : 0);
      return;
    }
  }
  fail("device request timed out");
}
} // namespace

extern "C" void rtl_init(void) {
  dut = std::make_unique<VKL_nvm_backend>();
  dut->rst_n = 0;
  dut->change_i = 0;
  dut->dev_req_i = dut->dev_wvalid_i = dut->dev_rready_i = 0;
  dut->mem_req_ready_i = dut->mem_rsp_valid_i = 0;
  dut->mem_rsp_data_i = 0;
  dut->mem_rsp_last_i = 1;
  dut->mem_rsp_err_i = 0;
  clear_csr();
  for (int n = 0; n < 8; ++n) edge();
  dut->rst_n = 1;
  edge();
}

extern "C" void rtl_reset_pulse(void) {
  log("reset_before");
  dut->rst_n = 0;
  for (int n = 0; n < 4; ++n) edge();
  dut->rst_n = 1;
  edge();
  log("reset_after");
}

extern "C" void rtl_csr_write(unsigned addr, uint32_t value) {
  // Translate the host pointer to a fixed, aligned simulation address only.
  // Geometry, length, map tables and all other control bits come from firmware.
  drive_csr(addr, addr == 0 ? kBase : value);
  if (addr == 4 && (value & 2)) log("firmware_ack_preedge");
  edge();
  clear_csr();
  if (addr == 4 && (value & 2)) {
    log("firmware_ack_postedge");
    std::printf("RTL ack_cycle=%llu preceding_write_done_cycle=%llu gap_cycles=%llu\n",
        (unsigned long long)ack_cycle, (unsigned long long)write_done_cycle,
        (unsigned long long)(ack_cycle - write_done_cycle));
  }
  if (addr == 4 && (value & 8)) log("firmware_arm_postedge");
}

extern "C" uint32_t rtl_status(void) { return status(); }
extern "C" uint64_t rtl_cycles(void) { return cycles; }
extern "C" void rtl_log(const char *tag) { log(tag); }
extern "C" void rtl_finish(void) { dut->final(); dut.reset(); }

extern "C" void rtl_record_erase(unsigned record_id) {
  log("erase_before");
  command(2, record_id, {}, false, false);
  log("erase_after");
}

extern "C" void rtl_record_program(unsigned image_offset, const uint8_t *data,
                                   unsigned len, int coincident_ack) {
  if (len < 8 || data[0] != 0x17 || data[1] != 0x22) fail("unframed record");
  const std::vector<uint8_t> frame(data, data + len);  // immutable stimulus copy
  log("program_before");
  command(1, frame[3], frame, coincident_ack != 0, false);
  if (std::memcmp(nvm_host_ddr + NVM_HOST_IMAGE_OFF + image_offset,
                  frame.data(), len) != 0)
    fail("backend output bytes differ from stimulus");
  log("program_after");
}

extern "C" void rtl_record_program_fail(unsigned record_id, const uint8_t *data,
                                        unsigned len) {
  const std::vector<uint8_t> frame(data, data + len);
  log("program_fail_before");
  fail_next_mem_write = true;
  command(1, record_id, frame, false, true);
  log("program_fail_after");
}

extern "C" void rtl_record_update(unsigned image_offset, const uint8_t *data,
                                  unsigned len, int coincident_ack) {
  if (len < 8 || data[0] != 0x17 || data[1] != 0x22) fail("unframed record");
  const unsigned first = bytes_committed;
  const unsigned ops = accepted;
  std::printf("RTL record_id=0x%02x framed_bytes=%u image_offset=%u start_cycle=%llu\n",
      data[3], len, image_offset, (unsigned long long)cycles);
  rtl_record_erase(data[3]);
  rtl_record_program(image_offset, data, len, coincident_ack);
  if (accepted - ops != 2 || bytes_committed - first != len * 2)
    fail("expected one full-region erase and one full-record program");
  std::printf("RTL completed_record_id=0x%02x device_ops=%u committed_bytes=%u write_done_cycle=%llu ack_cycle=%llu\n",
      data[3], accepted - ops, bytes_committed - first,
      (unsigned long long)write_done_cycle, (unsigned long long)ack_cycle);
}
