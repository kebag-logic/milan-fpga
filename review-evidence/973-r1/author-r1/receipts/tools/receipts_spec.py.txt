def unaddressed(text: str) -> str:
    """Host heap addresses differ run to run: replaced by <address>."""
    return ADDR.sub('<address>', text)


def shard(text: str) -> str:
    counts = (f"# [ok] lines {sum(l.startswith('[ok]') for l in text.splitlines())}, "
              f"ESCAPED {text.count('ESCAPED')}, suite verdicts PASS {text.count('verdict: PASS')}, "
              f"FAIL {text.count('verdict: FAIL')}\n")
    return counts + lines(r'^HEAD |^\[unit|^\[ok\] arm|^\[FAIL|ESCAPED|verdict:|test_ctrl_firmware: |mutants: |^END |'
                 r'link-pass-|srp-poll-extra|== .*checks:', text,
                 '# firmware gate shard summary lines (test_ctrl_firmware.py --require-rv32 --self-test --shard k/4 '
                 '--jobs 8); the whole log is kept in scratch\n')


def leg(text: str) -> str:
    return lines(r'^HEAD |^CLOCK|^STEPS|^SLICES|^JOINS|groups graded|refus|^END', text)
