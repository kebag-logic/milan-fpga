#!/usr/bin/env python3
"""receipts.py SPEC OUT: build the round's receipts directory OUT from SPEC (a Python file run with keep/put/lines/run
in scope), sanitising host paths, then write OUT/INDEX.md from the files present (path, bytes, sha256 prefix, first
line). A file over 200 KB is not copied: its sha256 and size go to OUT/not-copied.txt instead."""
import hashlib
import re
import shutil
import subprocess
import sys
from pathlib import Path

SPEC, OUT = Path(sys.argv[1]).resolve(), Path(sys.argv[2]).resolve()
S = Path('<scratch>')
L = Path('<lane>')
LIMIT = 200 * 1024
SUBS = [('<scratch>', '<scratch>'), ('<lane>', '<lane>'),
        ('<sdk>', '<sdk>'), ('<litex>', '<litex>'), ('<tools>', '<tools>'),
        ('<mgmt>', '<mgmt>'), ('<tmp>', '<tmp>'), ('<data>', '<data>'),
        ('<home>', '<home>'), ('<systmp>/', '<systmp>/')]
skipped = []


def clean(text: str) -> str:
    for a, b in SUBS:
        text = text.replace(a, b)
    return text


def put(rel: str, text: str) -> None:
    dest = OUT / rel
    dest.parent.mkdir(parents=True, exist_ok=True)
    data = clean(text).encode()
    if len(data) > LIMIT:
        skipped.append(f'{rel} {len(data)} {hashlib.sha256(data).hexdigest()}')
        return
    dest.write_bytes(data)


def keep(rel: str, src: Path, f=None) -> None:
    if not Path(src).is_file():
        skipped.append(f'{rel} MISSING {clean(str(src))}')
        return
    text = Path(src).read_text(errors='replace')
    put(rel, f(text) if f else text)


def lines(pattern: str, text: str, head: str = '') -> str:
    rx = re.compile(pattern)
    return head + ''.join(ln + '\n' for ln in text.splitlines() if rx.search(ln))


def run(*argv: str, cwd: Path = L) -> str:
    return subprocess.run(argv, cwd=cwd, capture_output=True, text=True).stdout


if OUT.exists():
    shutil.rmtree(OUT)
OUT.mkdir(parents=True)
exec(compile(SPEC.read_text(), str(SPEC), 'exec'))
if skipped:
    (OUT / 'not-copied.txt').write_text('# rel bytes sha256 (over 200 KB, or missing)\n' + '\n'.join(skipped) + '\n')
rows = []
for p in sorted(x for x in OUT.rglob('*') if x.is_file() and x.name != 'INDEX.md'):
    data = p.read_bytes()
    first = data.decode(errors='replace').splitlines()[0][:110] if data else ''
    first = first.replace('|', '\\|')
    rows.append(f'| `{p.relative_to(OUT).as_posix()}` | {len(data)} | `{hashlib.sha256(data).hexdigest()[:16]}` | {first} |')
(OUT / 'INDEX.md').write_text(
    '# Receipts index (#973 round 1)\n\nGenerated by receipts.py from the files present, after sanitising (host paths '
    'replaced by `<scratch>`, `<lane>`, `<sdk>`, `<litex>`, `<tools>`, `<mgmt>`, `<tmp>`, `<data>`, `<home>`). '
    'Each row: path, size, sha256 prefix, first line.\n\n| File | Bytes | sha256 | First line |\n|---|---:|---|---|\n'
    + '\n'.join(rows) + '\n')
print(f'{len(rows)} files, {len(skipped)} not copied')
