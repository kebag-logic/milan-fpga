#!/usr/bin/env bash
# launch_base.sh <name> <rev> <point> [run.py args...]: a scratch clone of <rev>, then the CPU budget harness as unit
# milan-943-a800-base-<name>; exit 1 if no CPU range is free
set -u
S=<scratch>; L=<lane>
N=$1; R=$2; P=$3; shift 3
C=$(python3 <mgmt>/2026-09-23/cpu_ranges.py pick) || exit 1
[ -n "$C" ] || exit 1
D=$S/base-$N
if [ ! -d "$D" ]; then
  git clone -q --shared --no-checkout $L $D && cd $D && git -c advice.detachedHead=false checkout -q --detach $R
  for d in gptp-processor protocol-processor third_party/lwSRP third_party/tsn-c-stack third_party/verilog-axis; do git config submodule.$d.url $L/$d; done
  git -c protocol.file.allow=always submodule update -q --init gptp-processor protocol-processor third_party/lwSRP third_party/tsn-c-stack third_party/verilog-axis
fi
systemd-run --user --unit=milan-943-a800-base-$N --collect -p MemoryMax=10G -p MemorySwapMax=0 -p WorkingDirectory=$D \
  /usr/bin/taskset -c "$C" /usr/bin/bash -c "$S/tools/budget.sh $D $P $S/budget/base-$N 8 $* > $S/budget/base-$N.log 2>&1; echo \$? > $S/budget/base-$N.rc"
echo "launched base-$N on $C"
