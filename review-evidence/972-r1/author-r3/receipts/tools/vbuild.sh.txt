#!/usr/bin/env bash
# vbuild.sh <clone> <stem> <directive> <work-dir>: the split image's Vivado build (the opt lanes' gbuild.sh): the recipe
# argv from <work-dir>/argv.txt, its flow flags with <directive> as the placement directive, the bench's --eth-port e2,
# through milan_soc.py --build into <work-dir>/build_<directive>, 8 threads; then the image identity and the signoff
# corners against #818's floor. Run inside vivado-slot.sh.
set -u
L=$1; STEM=$2; D=$3; W=$4; O=$W/build_$D; S=<scratch>
mkdir -p "$O"; date -Iseconds > "$O.start"
source $S/bin/env.sh
export HOME=<home> TSN_GEN_ROOT=<mgmt>/tools/tsn-gen-deca300c
export XILINXD_LICENSE_FILE=<home>/.Xilinx/Xilinx.lic
source <home>/Xilinx2/2026.1/Vivado/settings64.sh
command -v vivado || { echo 'no vivado on PATH'; echo 4 > "$O.rc"; exit 4; }
cd "$L/sw/litex" || { echo 3 > "$O.rc"; exit 3; }
mapfile -t ARGV < "$W/argv.txt"
echo "head $(git -C "$L" rev-parse HEAD); status $(git -C "$L" status --porcelain | tr '\n' ' ')"
sha256sum "$L/configs/$STEM.yaml" "$L/sw/builder/out/$STEM/ctrl_app.elf" "$L/sw/builder/out/$STEM/soc_params.json"
FLOW=(--synth-directive AreaOptimized_high --opt-directive ExploreArea --place-directive "$D")
echo "python3 milan_soc.py ${ARGV[*]} --eth-port e2 --entity-gen-dir $L/configs/generated/$STEM ${FLOW[*]} --no-compile-software --vivado-max-threads 8 --build --output-dir $O"
python3 milan_soc.py "${ARGV[@]}" --eth-port e2 --entity-gen-dir "$L/configs/generated/$STEM" "${FLOW[@]}" \
  --no-compile-software --vivado-max-threads 8 --build --output-dir "$O"
rc=$?
echo "$rc" > "$O.rc"; date -Iseconds > "$O.end"
echo "== END-OF-BUILD rc=$rc $(date -Iseconds)"
python3 "$L/sw/litex/check_image_identity.py" "$O" > "$O.identity.log" 2>&1; echo "identity rc=$?: $(tail -1 "$O.identity.log")"
python3 -I $S/bin/signoff.py "$O/gateware" > "$O.signoff.txt" 2>&1; echo "signoff rc=$?"; cat "$O.signoff.txt"
exit $rc
