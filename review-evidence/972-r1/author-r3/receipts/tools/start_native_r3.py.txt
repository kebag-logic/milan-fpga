#!/usr/bin/env python3
"""Launch the required four-shard native sweep only after routes and with four free CPU bookings."""
import json, subprocess, sys
from pathlib import Path
S=Path('<scratch>')
L=Path('<lane>')
M=Path('<mgmt>/2026-09-23')
assert (S/'units/viv-8x8-eto.rc').is_file(), 'last route is still running'
assert not (S/'native-r3').exists(), 'native sweep already started'
mem=int(next(x.split()[1] for x in Path('/proc/meminfo').read_text().splitlines() if x.startswith('MemAvailable:')))
if mem<=30*1024*1024: raise SystemExit('less than 30 GiB: defer parallel launch')
sys.path.insert(0,str(M)); import cpu_ranges
busy=cpu_ranges.busy(); cpus=[]
for _ in range(4):
    picked=cpu_ranges.pick8(busy)
    if not picked: raise SystemExit(75)
    cpus.append(picked); busy |= cpu_ranges.expand(picked)
head=subprocess.check_output(['git','-C',str(L),'rev-parse','HEAD'],text=True).strip()
base=subprocess.check_output(['git','-C',str(L),'rev-parse','dd163e41'],text=True).strip()
path='<litex>/venv/bin:<sdk>/host/bin:<tools>/pinned-verilator-5.050:<tools>/sv2v-v0.0.12:<data>/scratch/a662/make43/bin:<home>/.local/bin:/usr/local/sbin:/usr/local/bin:/usr/bin:/bin'
spec={'cwd':str(L),'head':head,'base':base,'commands':[['env','PATH='+path,
 'VERILATOR=<tools>/pinned-verilator-5.050/verilator',
 'TSN_GEN_ROOT=<mgmt>/tools/tsn-gen-deca300c',
 'TMPDIR='+str(S/'tmp'),'PYTHONHASHSEED=0','PYTHONDONTWRITEBYTECODE=1',
 'MAKEFLAGS=-j4 --no-print-directory','bash','scripts/run_all_suites.sh',str(S/'native-unused'),'--wait']]}
sp=S/'native-r3.json'; sp.write_text(json.dumps(spec,indent=2)+'\n')
subprocess.run(['python3',str(M/'native-shards.py'),'970-a820-r3native',str(sp),str(S/'native-r3'),'4',*cpus],check=True)
print('START native4',head,cpus)
