445	def d_clock_source(index: int, name: str, cs_type: int, loc_type: int,
446	                   loc_index: int) -> bytes:
447	    """CLOCK_SOURCE descriptor (7.2.9): one selectable media-clock reference.
448	
449	    `cs_type` is the 1722.1 encoding (INTERNAL=0 / INPUT_STREAM=2), which is
450	    why CRF and a stream clock are indistinguishable here and the config's own
451	    word has to travel separately - see clock_source_shape().
452	    """
453	    b = be16(CLOCK_SOURCE) + be16(index)
454	    b += cstr(name)                     # object_name
455	    b += be16(NO_STRING)
456	    b += be16(0x0002)                   # clock_source_flags (STREAM_ID)
457	    b += be16(cs_type)                  # INTERNAL=0 / INPUT_STREAM=2
458	    b += be64(0)                        # clock_source_identifier
459	    b += be16(loc_type)                 # location: descriptor holding the source
460	    b += be16(loc_index)
461	    assert len(b) == 86
462	    return b
