"""Enabled 8x8/1if route: firmware link, measurement export, synthesis, release placement.

Release recipe: AreaOptimized_high synthesis, ExploreArea opt, ExtraPostPlacementOpt place.
Only if that placement misses the #818 floor are AltSpreadLogic_high and ExtraTimingOpt
tried from the same checkpoint (best of three). Vivado runs one at a time through the
slot wrapper (VIVADO_SLOTS=2, VIVADO_MIN_GB=20).
"""

def event(**kw):
    kw['time'] = time.strftime('%Y-%m-%dT%H:%M:%S')
    log.write(json.dumps(kw) + '\n'); log.flush(); print(json.dumps(kw), flush=True)


def trees():
    out = {p: subprocess.check_output(['git', 'rev-parse', 'HEAD:' + p], cwd=root, text=True).strip()
           for p in ('hdl', 'sw', 'configs')}
    out['dirty_hdl_sw_configs'] = subprocess.check_output(
        ['git', 'status', '--porcelain', '--', 'hdl', 'sw', 'configs'], cwd=root, text=True)
    out['submodules'] = subprocess.check_output(['git', 'submodule', 'status'], cwd=root, text=True)
    return out

def vrun(cwd, tcl):
    c = [slot, vivado, '-mode', 'batch', '-nojournal', '-nolog', '-source', str(tcl)]
    start = time.time()
    with tcl.with_suffix('.log').open('w') as s:
        r = subprocess.run(c, cwd=cwd, env=env, stdout=s, stderr=subprocess.STDOUT, timeout=14400)
    res = {'command': c, 'cwd': str(cwd), 'exit': r.returncode, 'seconds': time.time() - start}
    tcl.with_suffix('.json').write_text(json.dumps(res, indent=2))
    event(step='vivado', tcl=str(tcl), exit=r.returncode, seconds=res['seconds'])
    return r.returncode


def timing(p):
    text = p.read_text()
    m = re.search(r'WNS\(ns\).*?\n[^\n]*\n\s*(-?\d+\.\d+)\s+(-?\d+\.\d+)\s+(\d+)\s+(\d+)\s+(-?\d+\.\d+)', text, re.S)
    values = text[m.start():].splitlines()[2].split()
    return {'WNS_ns': float(m[1]), 'TNS_ns': float(m[2]), 'WHS_ns': float(m[5]),
            'setup_failing_endpoints': int(values[2]), 'hold_failing_endpoints': int(values[6]),
            'sha256': hashlib.sha256(p.read_bytes()).hexdigest()}


def passes(d):
    corners = sorted(d.glob('*signoff*_*C_timing.rpt'))
    st = d / 'alinx_ax7101_route_status.rpt'
    if len(corners) != 4 or not st.exists():
        return False, None
    t = [timing(c) for c in corners]
    errs = int(re.search(r'nets with routing errors[^:]*:\s*(\d+)', st.read_text())[1])
    ok = (min(c['WNS_ns'] for c in t) >= .030 and min(c['WHS_ns'] for c in t) >= 0 and errs == 0
          and all(c['setup_failing_endpoints'] == 0 and c['hold_failing_endpoints'] == 0 for c in t))
    return ok, (min(c['WNS_ns'] for c in t), min(c['WHS_ns'] for c in t), errs)
