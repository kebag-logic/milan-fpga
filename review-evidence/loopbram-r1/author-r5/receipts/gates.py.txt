def run(case):
    name, cmd = case
    env = dict(os.environ, **(benv if group == 'builder' else {}))
    t = time.time()
    with (w / (name + '.log')).open('w') as f:
        r = subprocess.run(cmd, cwd=root, env=env, stdout=f, stderr=subprocess.STDOUT, timeout=43200)
    ok = r.returncode == 0
    if name == 'anchors':
        ok = ok and (w / 'anchors.log').read_text().rstrip().endswith('ANCHORS INTACT 0')
    res = dict(name=name, head=head, command=cmd, exit=r.returncode, pass_gate=ok, seconds=round(time.time() - t, 1))
    results.append(res)
    (w / (group + '.json')).write_text(json.dumps(results, indent=2) + '\n')
    print(name, 'PASS' if ok else 'FAIL', r.returncode, flush=True)
    return ok
