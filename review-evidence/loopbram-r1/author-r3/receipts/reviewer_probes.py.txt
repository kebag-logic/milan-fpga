"""Reviewer probes for the #979 input-mapped loopback wire harness.

Usage: probes.py --tree <exported head tree> --out <dir> --verilator <bin> [--pool N] [--jobs J]

Each probe copies tb/verilator/input_loopback and the two leaf sources into
its own directory, applies at most one textual plant per file (each plant must
match exactly once), builds with the pinned simulator and records the exit
status and the harness verdict line. Nothing is written into the tree.
"""

import json
import shutil

SHAPES = {8: dict(STREAMS=8, FIRST=1, CHANNELS=63, LEAD=1),
          1: dict(STREAMS=1, FIRST=8, CHANNELS=8, LEAD=2)}
Q, C, T = "queue", "capture", "sim"
TOL = ("localparam int STAMP_TOL_NS_C = 4 * STAMP_CYCLE_NS_C + STAMP_RATE_NS_C + 1;",
       "localparam int STAMP_TOL_NS_C = 0;")
STAMP = "if(events) d->frame_ns_i=stamp(frame)+uint32_t(phase>=250 ? capture_offset(frame) : 0);"


def drift(j):
    return (STAMP, f"if(events) d->frame_ns_i=stamp(frame)+uint32_t(phase>=250 ? {j} : 0);")


# name, shape, lead override, [(file, old, new)], expectation

def run(probe, tree, out, verilator, jobs):
    name, shape, lead, plants, expect = probe
    where = out / name
    if where.exists():
        shutil.rmtree(where)
    tb = where / "tb" / "verilator" / "input_loopback"
    shutil.copytree(tree / "tb/verilator/input_loopback", tb)
    shutil.copytree(tree / "tb/common", where / "tb" / "common")
    files = {Q: where / "KL_render_queue.sv", C: where / "KL_chan_map_capture.sv", T: tb / "sim_main.cpp"}
    shutil.copy(tree / "hdl/ieee1722/aaf/KL_render_queue.sv", files[Q])
    shutil.copy(tree / "hdl/ieee1722/aaf/KL_chan_map_capture.sv", files[C])
    for kind, old, new in plants:
        text = files[kind].read_text()
        if text.count(old) != 1:
            return dict(probe=name, error=f"plant text matched {text.count(old)} times in {kind}")
        files[kind].write_text(text.replace(old, new, 1))
    rtl = tree / "hdl"
    srcs = " ".join(str(p) for p in (
        rtl / "common/ethernet_packet_pkg.sv", rtl / "ieee1722/avtp/avtp_subtype_pkg.sv",
        rtl / "ieee1722/aaf/KL_aaf_packetizer.sv", rtl / "ieee1722/avtp/avtp_stream_parser.sv",
        rtl / "ieee1722/avtp/KL_avtp_rx_monitor.sv", rtl / "ieee1722/aaf/KL_aaf_rx_depacketizer.sv",
        files[Q], files[C], tree / "third_party/verilog-axis/rtl/axis_fifo.v", tb / "loopback_wrap.sv"))
    params = dict(SHAPES[shape])
    if lead is not None:
        params["LEAD"] = lead
    cmd = ["make", "-s", "-C", str(tb), "run", f"VERILATOR={verilator}", f"VERILATOR_JOBS={jobs}",
           f"MDIR={where / 'obj'}", f"RTL={rtl}", f"SRCS={srcs}"] + [f"{k}={v}" for k, v in params.items()]
    log = where / "run.log"
    start = time.time()
    with log.open("w") as sink:
        rc = subprocess.run(cmd, stdout=sink, stderr=subprocess.STDOUT, timeout=3000).returncode
    text = log.read_text()
    verdict = [l for l in text.splitlines() if "input-loopback streams=" in l]
    fails = sorted({l.split()[1] for l in text.splitlines() if l.startswith("FAIL ") and "input-loopback" not in l})
    built = (where / "obj" / "Vinput_loopback").exists()
    if expect == "pass":
        ok = rc == 0 and bool(verdict) and verdict[-1].startswith("PASS")
    elif expect == "fail":
        ok = built and rc != 0 and bool(verdict) and verdict[-1].startswith("FAIL")
    else:
        ok = None
    shutil.rmtree(where / "obj", ignore_errors=True)
    return dict(probe=name, shape=shape, params=params, plants=[[k, o, n] for k, o, n in plants],
                expect=expect, exit=rc, built=built, verdict=verdict[-1] if verdict else None,
                failing_labels=fails, expectation_met=ok, seconds=round(time.time() - start, 1))


def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--tree", type=Path, required=True)
    ap.add_argument("--out", type=Path, required=True)
    ap.add_argument("--verilator", required=True)
    ap.add_argument("--pool", type=int, default=6)
    ap.add_argument("--jobs", type=int, default=2)
    a = ap.parse_args()
    a.out.mkdir(parents=True, exist_ok=True)
    results = []
    with cf.ThreadPoolExecutor(a.pool) as ex:
        futs = [ex.submit(run, p, a.tree.resolve(), a.out.resolve(), a.verilator, a.jobs) for p in PROBES]
        for f in cf.as_completed(futs):
            r = f.result()
            results.append(r)
            print(json.dumps(r), flush=True)
            (a.out / "results.json").write_text(json.dumps(sorted(results, key=lambda r: r["probe"]), indent=1) + "\n")
    bad = [r for r in results if r.get("error") or r.get("expectation_met") is False]
    print("PROBES DONE", len(results), "unexpected", [r["probe"] for r in bad], flush=True)
    raise SystemExit(1 if bad else 0)


if __name__ == "__main__":
    main()
