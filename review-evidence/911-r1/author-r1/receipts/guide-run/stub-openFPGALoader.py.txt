#!/usr/bin/env python3
"""Stub programmer for the guide run: never opens a cable. --detect prints the die STUB_MODEL (default xc7a200) and,
with -f, the flash ID STUB_JEDEC (default 20ba18) the way openFPGALoader --verbose does; writes and dumps act on the
file STUB_QSPI (a 16 MiB image) and every call is appended to STUB_LOG."""
import os, struct, sys
args = sys.argv[1:]
with open(os.environ["STUB_LOG"], "a", encoding="utf-8") as log:
    log.write(" ".join(args) + "\n")
def value(flag, default=None):
    return args[args.index(flag) + 1] if flag in args else default
if "--detect" in args:
    print("index 0:\n\tidcode 0x3636093\n\tmanufacturer xilinx\n\tfamily artix a7\n\tmodel  "
          + os.environ.get("STUB_MODEL", "xc7a200") + "\n\tirlength 6")
    jedec = os.environ.get("STUB_JEDEC", "20ba18")
    if "-f" in args and jedec:
        print(" ".join(f"{b:x}" for b in bytes.fromhex(jedec)) + f" read {int(jedec, 16):x}")
    sys.exit(0)
qspi = os.environ["STUB_QSPI"]
if not os.path.exists(qspi):
    with open(qspi, "wb") as f:
        f.truncate(16 << 20)
if "--dump-flash" in args:
    with open(qspi, "rb") as f, open(args[-1], "wb") as out:
        f.seek(int(value("-o", "0"), 0)); out.write(f.read(int(value("--file-size"), 0)))
    sys.exit(0)
raw = open(args[-1], "rb").read()
if "-f" in args and value("-o") is None:
    pos = 2 + struct.unpack_from(">H", raw, 0)[0] + 2
    while raw[pos] != ord("e"):
        pos += 3 + struct.unpack_from(">H", raw, pos + 1)[0]
    raw = raw[pos + 5:pos + 5 + struct.unpack_from(">I", raw, pos + 1)[0]]
with open(qspi, "r+b") as f:
    f.seek(int(value("-o", "0"), 0)); f.write(raw)
