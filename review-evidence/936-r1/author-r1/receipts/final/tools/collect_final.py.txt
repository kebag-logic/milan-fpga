#!/usr/bin/env python3
"""collect_final.py: the final-head receipts of #936 round 1 into <output>/receipts/final (sanitized: no host paths,
full SHAs cut to 8; a file over 180 KB becomes an excerpt with its original size and sha256), the earlier final/
content moved to receipts/earlier/, then receipts/INDEX.md generated from the files present."""
import hashlib, json, re, shutil, subprocess, sys
from pathlib import Path

S = Path('<scratch>')
O = Path('<output>')
R = O / 'receipts'
F = R / 'final'
LANE = '<lane>'
LIMIT = 180_000
PATHS = [(str(O), '<output>'), (str(S), '<scratch>'), (LANE, '<lane>'), ('<tmp>', '<tmp>'),
         ('<home>', '<home>'), ('<workspace>', '<workspace>'), ('<workspace>', '<workspace>')]


def sanitize(s: str) -> str:
    for a, b in PATHS:
        s = s.replace(a, b)
    return re.sub(r'(?<![0-9a-f])[0-9a-f]{40}(?![0-9a-f])', lambda m: m[0][:8], s)


def put(rel: str, text: str) -> None:
    d = F / rel
    d.parent.mkdir(parents=True, exist_ok=True)
    text = sanitize(text)
    assert len(text.encode()) <= 200_000, (rel, len(text.encode()))
    d.write_text(text)


KEEP = ('plant ', '[DIFF]', 'TWIN', '[OWED]', '[LEDGER]', 'twin:', 'RESULT', 'PASS', 'FAIL', '[ok]', '[ESCAPED]',
        'mutants:', 'suites:', 'checks:', 'test_ctrl_firmware', 'firmware coverage', 'REFUSED', 'BUDGET', 'SHARE',
        'STATED', 'groups graded', 'elapsed=', 'HEAD ', '# head', 'shard:', 'saved-state')


def take(src: Path, rel: str) -> None:
    data = src.read_bytes()
    text = data.decode(errors='replace')
    head = f'Original file: {sanitize(str(src))}\nOriginal size: {len(data)} bytes\nOriginal sha256: ' \
           f'{hashlib.sha256(data).hexdigest()}\n\n'
    if len(data) > LIMIT:
        lines = text.splitlines()
        kept = [l[:400] for l in lines if l.lstrip().startswith(KEEP)]
        body = 'EXCERPT ONLY (the file is over 180 KB; the original is kept in scratch, its sha256 above)\n' + \
               '\n'.join(kept[:600]) + '\n--- last 20 lines ---\n' + '\n'.join(l[:400] for l in lines[-20:]) + '\n'
        while len(sanitize(head + body).encode()) > 195_000:
            kept = kept[:len(kept) * 3 // 4]
            body = 'EXCERPT ONLY (the file is over 180 KB; the original is kept in scratch, its sha256 above)\n' + \
                   '\n'.join(kept) + '\n--- last 20 lines ---\n' + '\n'.join(l[:400] for l in lines[-20:]) + '\n'
        put(rel, head + body)
    else:
        put(rel, head + text)


def git(*a: str) -> str:
    return subprocess.run(['git', *a], cwd=LANE, check=True, text=True, capture_output=True).stdout
