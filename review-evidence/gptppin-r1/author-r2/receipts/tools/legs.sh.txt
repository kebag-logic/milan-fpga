#!/usr/bin/env bash
# legs.sh <clone> <outdir> <suite:target>...: each named make target of a suite on a clean clone, the pinned
# Verilator 5.050 first on PATH, all targets at once, each under timeout LEG_TIMEOUT (default 590 s) and
# scripts/suite_tally.py's verdict; one line per target in <outdir>/SUMMARY.txt.
set -u
C=$1; O=$2; shift 2
export PATH=<tools>/pinned-verilator-5.050:<tools>/sv2v-v0.0.12:$PATH
export VERILATOR=<tools>/pinned-verilator-5.050/verilator MAKEFLAGS=--no-print-directory PYTHONHASHSEED=0
mkdir -p "$O" "$O/tmp"; export TMPDIR="$O/tmp"
echo "head $(git -C "$C" rev-parse HEAD) porcelain $(git -C "$C" status --porcelain --untracked-files=no | wc -l)" > "$O/SUMMARY.txt"
one() {
  local st=$1 s t n t0 rc
  s=${st%%:*}; t=${st#*:}; [ "$t" = "$st" ] && t=""
  n=$s${t:+-$t}; t0=$(date +%s)
  timeout ${LEG_TIMEOUT:-590} make -C "$C/tb/verilator/$s" $t > "$O/$n.log" 2>&1; rc=$?
  if [ "$rc" -eq 0 ] && ! python3 "$C/scripts/suite_tally.py" --verdict "$O/$n.log" > "$O/$n.verdict" 2>&1; then rc=93; fi
  echo "$n rc=$rc seconds=$(( $(date +%s) - t0 ))" >> "$O/SUMMARY.txt"
}
for st in "$@"; do one "$st" & done
wait
echo "porcelain-after $(git -C "$C" status --porcelain --untracked-files=no | wc -l)" >> "$O/SUMMARY.txt"
echo DONE >> "$O/SUMMARY.txt"
