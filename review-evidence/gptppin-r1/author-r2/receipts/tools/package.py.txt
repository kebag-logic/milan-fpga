def clean(text: str) -> str:
    for a, b in SUBS:
        text = text.replace(a, b)
    return text


def put(dest: str, text: str) -> None:
    path = O / dest
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(clean(text))
    assert path.stat().st_size < 200_000, (dest, path.stat().st_size)


for dest, src, head in FILES:
    if src is None:
        continue
    put(dest, (head + "\n" if head else "") + (S / src).read_text(errors="replace"))
