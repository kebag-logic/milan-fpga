#!/usr/bin/env python3
"""docs_steps.py <head> <root> [job ...]: every `run:` step of docs.yml's gate jobs, read from the clone's own
workflow, on one fresh clean clone of <head>; each step's log and rc under <root>/logs, a summary line per step.

Not run, each with its reason (printed as SKIP): tool installs and package resolution (the host carries the pinned
tools: the markdown venv on PATH, sv2v v0.0.12, the RV32 SDK at $HOME/br-milan-rv32/host); the hosted runner's
evidence recorders and fail-fast steps, which need the run's artifacts. `python3 -m pip install` lines inside a gate
step are dropped (the venv has the packages). The em-dash gate's base is BASE (the flip train's head, which the PR's
base will carry once the train merges) in place of the event's fetch. docs-check-no-git runs on a copy of the clone
with its .git removed, as the job does.
"""
import os, re, shutil, subprocess, sys
from pathlib import Path
import yaml

LANE = "<lane>"
BASE = "15168e14a1"
VENV = "<tools>/md-venv-40cdefe08ebd/bin"
TOOLS = ("<home>/br-milan-rv32/host/bin:<tools>/pinned-verilator-5.050:<tools>/sv2v-v0.0.12:"
         "<tools>/a662/make43/bin:/usr/local/sbin:/usr/local/bin:/usr/bin:/bin")
JOBS = ("wire-accountability", "docs-check-no-git", "docs-gates-0", "docs-gates-1", "docs-gates-2")
SKIP = (("Install the python gate dependencies", "pip install of packages the venv carries"),
        ("Install the pinned Markdown renderer", "the venv carries the pinned renderer"),
        ("Resolve exact package downloads", "apt package resolution; the host has librsvg2's rsvg-convert"),
        ("Install diagram gate dependencies", "apt install; the host has the packages"),
        ("Install the pinned sv2v release", "the host carries sv2v v0.0.12 on PATH"),
        ("Install the pinned evidence reader", "the hosted evidence reader"),
        ("Record ", "the hosted runner's evidence recorders need the run's artifacts"),
        ("Retain checkout identity", "a hosted evidence recorder"),
        ("Collect actual runtime evidence", "needs the hosted run's artifacts"),
        ("Observe the first failed job", "hosted fail-fast"), ("Cancel ", "hosted fail-fast"))

head, root = sys.argv[1], Path(sys.argv[2])
jobs = sys.argv[3:] or JOBS
logs = root / "logs"
logs.mkdir(parents=True, exist_ok=True)
wt = root / "wt"
assert not wt.exists(), "a fresh clone"
run = lambda *a, cwd=None: subprocess.run(a, cwd=cwd, check=True, capture_output=True, text=True).stdout.strip()
run("git", "clone", "-q", "--shared", "--no-checkout", LANE, str(wt))
run("git", "-c", "advice.detachedHead=false", "checkout", "-q", "--detach", head, cwd=wt)
gm = run("git", "config", "--blob", f"{head}:.gitmodules", "--get-regexp", r"^submodule\..*\.path$", cwd=wt)
for line in gm.splitlines():
    dep = line.split()[1]
    if (Path(LANE) / dep / ".git").exists():
        run("git", "config", f"submodule.{dep}.url", f"{LANE}/{dep}", cwd=wt)
temp = root / "runner_temp"; temp.mkdir(exist_ok=True)
tmp = root / "tmp"; tmp.mkdir(exist_ok=True)
gh_env = root / "github_env"; gh_env.touch()
env = {k: v for k, v in os.environ.items() if k in ("LANG", "LC_ALL", "USER", "TERM")}
env.update(PATH=f"{VENV}:{TOOLS}", HOME="<home>", RUNNER_TEMP=str(temp), TMPDIR=str(tmp), GITHUB_WORKSPACE=str(wt),
           GITHUB_ENV=str(gh_env), GITHUB_OUTPUT=str(root / "github_output"), GITHUB_PATH=str(root / "github_path"),
           GITHUB_SHA=head, GITHUB_EVENT_NAME="pull_request", GITHUB_BASE_REF="dev-soft-core", CI="true",
           GIT_ALLOW_PROTOCOL="file:https", VERILATOR="<tools>/pinned-verilator-5.050/verilator",
           TSN_GEN_ROOT="<home>/milan-fpga-management/tools/tsn-gen-deca300c", PYTHONHASHSEED="0",
           LITEX_ENV_CC_TRIPLE="riscv32-buildroot-linux-gnu", COURSIER_MODE="offline", SBT_OPTS="-Dsbt.offline=true")
EVENT = {"github.workspace": str(wt), "github.event_name": "pull_request",
         "github.event.pull_request.base.ref": "dev-soft-core", "github.event.pull_request.head.sha": head,
         "github.event.before": "", "github.event.pull_request.draft": "false"}
expand = lambda t: re.sub(r"\$\{\{\s*([\w.-]+)\s*\}\}", lambda m: EVENT.get(m[1], m[0]), t)
wf = yaml.safe_load((wt / ".github/workflows/docs.yml").read_text())
n = 0
failed = 0
for job in jobs:
    body = wf["jobs"][job]
    for step in body["steps"]:
        if "run" not in step:
            continue
        name = step.get("name") or step.get("id") or ""
        n += 1
        tag = f"{n:02d}-{job}-{re.sub(r'[^A-Za-z0-9]+', '_', name)[:48]}"
        why = next((r for p, r in SKIP if name.startswith(p)), None)
        if why:
            print(f"{tag}\tSKIP\t{why}", flush=True)
            continue
        e = dict(env)
        for line in gh_env.read_text().splitlines():
            if "=" in line:
                k, v = line.split("=", 1)
                e[k] = v
        for k, v in {**(body.get("env") or {}), **(step.get("env") or {})}.items():
            if k != "PIP_CONSTRAINT":
                e[k] = expand(str(v))
        kept, dropping = [], False
        for l in expand(step["run"]).splitlines():
            if l.strip().startswith("python3 -m pip install"):
                dropping = l.rstrip().endswith("\\")
                continue
            if dropping:
                dropping = l.rstrip().endswith("\\")
                continue
            kept.append(l)
        text = "\n".join(kept)
        if name.startswith("Added-line em-dash gate"):
            text = f"python3 scripts/check_em_dash.py --base {BASE}"
        cwd = wt
        if job == "docs-check-no-git" and name.startswith("Strip git metadata"):
            cwd = root / "nogit"
            if cwd.exists():
                shutil.rmtree(cwd)
            shutil.copytree(wt, cwd, symlinks=True, ignore=shutil.ignore_patterns(".git"))
            text = "\n".join(l for l in text.splitlines() if l.strip() != "rm -rf .git")
        script = root / f"{tag}.sh"
        script.write_text(text + "\n")
        log = logs / f"{tag}.log"
        with open(log, "w") as out:
            out.write(f"# head {head} job {job} step: {name}\n")
            out.flush()
            rc = subprocess.run(["bash", "-e", str(script)], cwd=cwd / step.get("working-directory", "."), env=e,
                                stdout=out, stderr=subprocess.STDOUT, stdin=subprocess.DEVNULL).returncode
        if rc == 0 and "git submodule update" in text:
            # the hosted runner clones each submodule from its .gitmodules URL, so that URL is its origin (the builder
            # legs' source check reads it); the bytes here come from the lane's copy, checked against the gitlink
            with open(log, "a") as out:
                for line in gm.splitlines():
                    key, dep = line.split()
                    if (wt / dep / ".git").exists():
                        url = run("git", "config", "--blob", f"{head}:.gitmodules", key.replace(".path", ".url"), cwd=wt)
                        run("git", "remote", "set-url", "origin", url, cwd=wt / dep)
                        out.write(f"# origin restored: {dep} -> {url}\n")
        last = [l for l in log.read_text(errors="replace").splitlines() if l.strip()][-1:]
        print(f"{tag}\trc={rc}\t{last[0][:160] if last else ''}", flush=True)
        failed += rc != 0
print(f"tree porcelain: {len(run('git', 'status', '--porcelain', cwd=wt).splitlines())}")
print(f"DONE failed {failed}")
