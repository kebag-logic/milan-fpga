"""Step 2 against step 1: every census input that reaches the native run, byte for byte.

Compares, per variant: the native executable, the firmware ELF, the flash image, every recorded build input except
the two source-tree digests (which name the checkout, not the run), and per case the stimulus script the runner
would write now against the one step 1 ran. Prints IDENTICAL or each difference; exit 1 on any difference.
"""
import hashlib, json, sys
from pathlib import Path
root = Path.cwd()
sys.path.insert(0, str(root / 'tb/verilator/ctrl_service_budget'))
import build, run, scenario
NEW = Path('$VALIDATION_STORAGE/r7/final-census'); OLD = Path('$VALIDATION_STORAGE/r7/step1/final-census')
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
import re, subprocess, tempfile
NORMALIZED = ('build:gateware/sim.v', 'build:split-soc/flash-map/flash_port.json', 'build:runtime/libc.a',
              'build:runtime/libcompiler_rt.a')
DATED = re.compile(r'(Date\s*:|Auto-Generated by LiteX on|generated)', re.I)
def undated(path):
    return hashlib.sha256(''.join(l for l in Path(path).read_text(errors='replace').splitlines(True)
                                  if not DATED.search(l)).encode()).hexdigest()
def members(path):
    with tempfile.TemporaryDirectory(dir='$VALIDATION_STORAGE/r7/work-tmp') as tmp:
        subprocess.run(['ar', 'x', str(Path(path).resolve())], cwd=tmp, check=True)
        return sorted((m.name, sha(m)) for m in Path(tmp).iterdir())
def normalized(key, d):
    rel = key.split(':', 1)[1]
    if rel.endswith('.a'):
        return members(d / rel)
    if rel.endswith('sim.v'):
        return undated(d / rel)
    data = json.loads((d / rel).read_text())
    # The map records the hashes of the CSR files it was generated from; compare those files with dates removed.
    csr = d / 'split-soc/stand-ins/soc'
    data['source'] = {k: undated(csr / ('software/include/generated/csr.h' if k == 'csr.h' else k))
                      for k in data['source']}
    return data
diffs, report = [], {}
manifest = json.loads((OLD / 'manifest.json').read_text())
for variant, wanted in manifest['groups'].items():
    a, b = json.loads((OLD / variant / 'build.json').read_text()), build.reused(NEW / variant)
    row = dict(executable=(a['executable_sha256'], b['executable_sha256']), elf=(sha(a['elf']), sha(b['elf'])),
               flash=(a.get('flash_image') and sha(a['flash_image']), b.get('flash_image') and sha(b['flash_image'])))
    ia, ib = a.get('inputs', {}), b.get('inputs', {})
    for key in sorted(set(ia) | set(ib)):
        if key in ('tree:firmware', 'tree:config'):
            continue
        x, y = ia.get(key), ib.get(key)
        if x != y and key in NORMALIZED:
            # A generation timestamp or archive member date only: compared with it removed (normalized()).
            x, y = normalized(key, OLD / variant), normalized(key, NEW / variant)
            row['normalized ' + key] = (x, y)
            continue
        row['input ' + key] = (x, y)
    groups = {g.name: g for g in run.all_groups(scenario.point('arty-1x1', b), b)}
    for name in wanted:
        row['stimulus ' + name] = (sha(OLD / variant / 'results' / (name + '.txt')),
                                   hashlib.sha256(groups[name].script().encode()).hexdigest())
    for key, (x, y) in row.items():
        if x != y:
            diffs.append(f'{variant}: {key} differs: {x} -> {y}')
    report[variant] = {k: v[1] for k, v in row.items() if not k.startswith(('input ', 'stimulus ', 'normalized '))} | \
        {'inputs_compared': sum(k.startswith('input ') for k in row), 'stimuli_compared': sum(k.startswith('stimulus ') for k in row),
        'normalized_compared': sorted(k for k in row if k.startswith('normalized '))}
print(json.dumps(report, indent=1))
for line in diffs:
    print('DIFFERS', line)
print('IDENTICAL' if not diffs else f'{len(diffs)} DIFFERENCES')
raise SystemExit(bool(diffs))
