#!/usr/bin/env bash
# headsims2.sh: the split boot simulation at 1x1/1if and 8x8/1if, the platform ledger's self-test and the boot's
# self-test, on the clean clone at the round-2 head, one after another; each rc in <build-dir>.rc
set -u
S=<scratch>
C=$S/c-head2
bash $S/scripts/boot.sh $C endstation_ax7101_1x1_tdm8 $S/r3-boot-1x1 > $S/r3-boot-1x1.log 2>&1
bash $S/scripts/boot.sh $C endstation_ax7101_8x8 $S/r3-boot-8x8 > $S/r3-boot-8x8.log 2>&1
bash $S/scripts/ledger.sh $C $S/r3-ledger --self-test > $S/r3-ledger.log 2>&1
bash $S/scripts/boot.sh $C endstation_ax7101_1x1_tdm8 $S/r3-bootself --self-test > $S/r3-bootself.log 2>&1
echo "HEADSIMS3 DONE $(cat $S/r3-boot-1x1.rc $S/r3-boot-8x8.rc $S/r3-ledger.rc $S/r3-bootself.rc | tr '\n' ' ')"
