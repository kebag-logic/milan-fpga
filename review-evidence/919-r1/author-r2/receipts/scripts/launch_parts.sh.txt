#!/usr/bin/env bash
# launch_parts.sh: start each harness part not yet started on a CPU range cpu_ranges.py picks now (8 wide); print
# what it started and what still waits. A part is started at most once (its marker file).
S=<scratch>
for spec in h2-8x8-1if:8x8-1if:0 h2-8x8-1if:8x8-1if:1 h2-8x8-1if:8x8-1if:2 h2-8x8-1if:8x8-1if:3 h2-1x1:1x1:0 h2-1x1:1x1:1; do
  IFS=: read -r d p k <<< "$spec"
  [ -e "$S/$d.part$k.started" ] && continue
  # 8x8/1if's parts first: a 1x1 part takes a range only once every 8x8/1if part has one
  if [ "$p" = 1x1 ] && [ "$(ls $S/h2-8x8-1if.part*.started 2>/dev/null | wc -l)" -lt 4 ]; then echo "waiting $d part$k: 8x8/1if first"; continue; fi
  C=$(python3 <management>/2026-09-23/cpu_ranges.py pick 2>/dev/null | tr -d '[:space:]')
  if [ -z "$C" ]; then echo "waiting $d part$k: no range free"; continue; fi
  systemd-run --user --unit=milan-919-a785-h2-$p-part$k -p MemoryMax=8G -E HOME=<home> /usr/bin/taskset -c "$C" \
    /bin/bash -c "$S/scripts/budget2.sh $S/c-budget2 $p $S/$d 6 part$k --reuse-build --groups $(cat $S/$d.part$k) --out $S/$d/results > $S/$d.part$k.log 2>&1" \
    && echo "$C" > "$S/$d.part$k.started" && echo "started $d part$k on $C"
  sleep 2
done
