#!/usr/bin/env python3
"""Plant the control campaign's PHY defects (#919) alone, through ctrl_mutants.campaign, on one tree."""
import sys
from pathlib import Path
root = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(root / 'sw/firmware/ctrl/test'))
import ctrl_mutants  # noqa: E402
from ctrl_build import stack_pin  # noqa: E402
out = Path(sys.argv[2]).resolve()
stack_pin()
names = sys.argv[3].split(',') if len(sys.argv) > 3 else None
ctrl_mutants.MUTANTS = tuple(m for m in ctrl_mutants.MUTANTS if m.name.startswith(('plat-phy-', 'plat-link-', 'plat-store-step', 'plat-map-')) and (names is None or m.name in names))
print(f'{len(ctrl_mutants.MUTANTS)} mutants')
sys.exit(1 if ctrl_mutants.campaign(out / 'm', out / 'reuse', 4) else 0)
