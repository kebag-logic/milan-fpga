#!/usr/bin/env python3
"""srp8x8.py <tree-root> <build-dir> [suite ...]: run each SRP suite (default srp_mbx.cpp) through srp_arms.arm_srp at
both interface counts and at both shipped AX7101 entities, the arms' own recipe; print each outcome's tally and every
failing test, rc 1 when any arm failed."""
import sys
from pathlib import Path

root, out = Path(sys.argv[1]).resolve(), Path(sys.argv[2]).resolve()
suites = sys.argv[3:] or ["srp_mbx.cpp"]
sys.path.insert(0, str(root / "sw/firmware/ctrl/test"))
sys.path.insert(0, str(root / "sw/firmware/gtest"))
import fw_gtest  # noqa: E402
from ctrl_build import Tree, CTRL  # noqa: E402
from ctrl_reuse import cut_reuse  # noqa: E402
import srp_arms  # noqa: E402

tree = Tree(CTRL, out / "checkout", out / "reuse", fw_gtest.Build(jobs=4))
cut_reuse(tree.reuse)
lw = root / "third_party/lwSRP"
bad = 0
for suite in suites:
    for config in ("endstation_ax7101_1x1_tdm8.yaml", "endstation_ax7101_8x8.yaml"):
        for i in (1, 2):
            o = srp_arms.arm_srp(tree, lw, i, test=suite, config=root / "configs" / config)
            lines = o.log.splitlines()
            tally = [l for l in lines if l.startswith("[==========]") or l.startswith("[  PASSED  ]")
                     or l.startswith("[  FAILED  ]")]
            print(f"== {o.arm}: rc {o.rc}")
            for l in tally:
                print("   " + l)
            bad |= bool(o.rc)
sys.exit(1 if bad else 0)
