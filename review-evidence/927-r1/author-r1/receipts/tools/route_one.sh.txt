#!/usr/bin/env bash
# route_one.sh <gateware dir>: the integrated route (S9's recipe) in an exported gateware dir; writes route.rc,
# route.start, route.end there.
set -u
g=$1
source <tools>/vivado/Vivado/settings64.sh
cd "$g" || exit 3
date -Iseconds > route.start
vivado -mode batch -source baseline_integrated.tcl -nojournal -log baseline.log > vivado.out 2>&1
echo $? > route.rc
date -Iseconds > route.end
echo "$g rc=$(cat route.rc)"
