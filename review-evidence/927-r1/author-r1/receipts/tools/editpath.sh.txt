#!/usr/bin/env bash
# editpath.sh <clone> <out>: every simulation and co-simulation suite on the edit path, one make target each, on a
# clean clone: the split leg and its plants, the split crossing's structure, the map-edit legs and their plants, the
# two-placement simulation and its plants, the mailbox suite, the AAF clock meter and its mutants. rc per target in
# <out>/<suite>__<target>.rc, log in .log
set -u
C=$1; O=$2; mkdir -p "$O"
export PATH=<tools>/pinned-verilator-5.050:<tools>/sv2v-v0.0.12:<milan>/scratch/a662/make43/bin:/usr/local/bin:/usr/bin:/bin:<tools>/rv32/host/bin
export VERILATOR=<tools>/pinned-verilator-5.050/verilator SV2V=<tools>/sv2v-v0.0.12/sv2v YOSYS=/usr/bin/yosys
export TSN_GEN_ROOT=<management>/tools/tsn-gen-deca300c PYTHONHASHSEED=0
export TMPDIR=<scratch>/tmp/editpath HOME=<scratch>/wf/home COURSIER_MODE=offline SBT_OPTS=-Dsbt.offline=true
mkdir -p "$TMPDIR"
cd "$C" || exit 3
echo "head $(git rev-parse HEAD) dirty $(git status --porcelain --untracked-files=no | wc -l) start $(date -Iseconds)" | tee "$O/RUN.txt"
run() { local s=$1 t=$2; local st=$(date +%s)
  make -C "tb/verilator/$s" -j16 --no-print-directory $t > "$O/${s}__${t}.log" 2>&1; local rc=$?
  echo $rc > "$O/${s}__${t}.rc"; echo "$(date +%T) $s $t rc=$rc $(( $(date +%s) - st ))s" | tee -a "$O/RUN.txt"; }
run milan_dp split
run milan_dp split-mutants
run milan_dp split-cdc
run milan_dp dynmap
run milan_dp maprec8
run milan_dp maprec4
run milan_dp dynmap-mutants
run milan_dp twin
run milan_dp twin-plants
run mbx all
run aaf_clock_meter all
echo "porcelain-after $(git status --porcelain --untracked-files=no | wc -l) end $(date -Iseconds)" | tee -a "$O/RUN.txt"
