#!/usr/bin/env bash
# unitpool.sh <name> <memmax> <log> <cmd...>: as unit.sh, but on the shared session pool (cpu_ranges.SESSION_POOL,
# 96-127: shared, never a booking), for campaigns and single legs while no booked range is free.
set -u
N=$1; M=$2; L=$3; shift 3
C=$(python3 -I -c "import sys; sys.path.insert(0, '<management>/2026-09-23'); import cpu_ranges; print(cpu_ranges.session_pool())")
systemd-run --user --unit="milan-927a791-$N" --collect -p MemoryMax="$M" -p MemorySwapMax=0 \
  --setenv=HOME=<home> --setenv=PATH=/usr/local/sbin:/usr/local/bin:/usr/bin:/bin \
  /usr/bin/bash -c 'exec /usr/bin/taskset -c "$0" "$@" > "'"$L"'" 2>&1' "$C" "$@"
echo "unit milan-927a791-$N cpus $C (session pool) log $L"
