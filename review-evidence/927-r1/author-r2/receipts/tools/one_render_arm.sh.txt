#!/usr/bin/env bash
# one_render_arm.sh <clone> <mutants.py> <name>...: tdm8_render_mutants.py's run_mutations for the named arms only, in
# <clone>'s milan_dp_render with <mutants.py> copied over the clone's copy (restored after), editrun.sh's environment.
# A quick check that an arm is caught before the whole target runs on a clean clone; not evidence on its own.
set -u
C=$1; M=$2; shift 2
export PATH=<tools>/pinned-verilator-5.050:<tools>/sv2v-v0.0.12:<milan>/scratch/a662/make43/bin:/usr/local/bin:/usr/bin:/bin:<tools>/rv32/host/bin
export VERILATOR=<tools>/pinned-verilator-5.050/verilator SV2V=<tools>/sv2v-v0.0.12/sv2v YOSYS=/usr/bin/yosys
export TSN_GEN_ROOT=<management>/tools/tsn-gen-deca300c PYTHONHASHSEED=0 PYTHONUNBUFFERED=1
export TMPDIR=<scratch>/tmp/one-arm HOME=<scratch>/wf/home COURSIER_MODE=offline SBT_OPTS=-Dsbt.offline=true
mkdir -p "$TMPDIR"; D=$C/tb/verilator/milan_dp_render
cp "$M" "$D/tdm8_render_mutants.py"
cd "$D" && python3 - "$@" <<'EOF'
import sys, tempfile
from pathlib import Path
sys.path.insert(0, '.')
import tdm8_render_mutants as m
names = sys.argv[1:]
m.MUTATIONS = [a for a in m.MUTATIONS if a[0] in names]
assert len(m.MUTATIONS) == len(names), [a[0] for a in m.MUTATIONS]
_run = m.run_leg
def run_leg(exe, mode):
    rc, out = _run(exe, mode)
    fails = [l for l in out.splitlines() if l.strip().startswith('[FAIL]')]
    print(f'leg rc {rc}; {len(fails)} [FAIL] line(s):')
    for l in fails[:40]:
        print('   ', l[:300])
    return rc, out
m.run_leg = run_leg
with tempfile.TemporaryDirectory(prefix='one-arm-') as td:
    p, f = m.run_mutations(Path(td))
print(f'arms: {p} caught, {f} failed')
sys.exit(1 if f else 0)
EOF
rc=$?
git -C "$C" checkout -q -- tb/verilator/milan_dp_render/tdm8_render_mutants.py
echo "rc=$rc restored $(git -C "$C" status --porcelain --untracked-files=no | wc -l)"
exit $rc
