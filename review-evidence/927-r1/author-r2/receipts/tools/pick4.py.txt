"""pick4.py: four distinct 8-wide ranges from cpu_ranges.pick8, each taken before the next is picked (pick8 reads the
live bookings, so four calls before any unit starts would return the same range). Prints them space-separated and
MemAvailable in GB; exits 1, printing nothing, when fewer than four are free or MemAvailable is 30 GB or less."""
import sys
sys.path.insert(0, '<management>/2026-09-23')
import cpu_ranges  # noqa: E402

avail = int(next(l for l in open('/proc/meminfo') if l.startswith('MemAvailable')).split()[1]) // 1048576
taken = cpu_ranges.busy()
picks = []
for _ in range(4):
    r = cpu_ranges.pick8(taken)
    if not r:
        break
    picks.append(r)
    taken |= cpu_ranges.expand(r)
if len(picks) < 4 or avail <= 30:
    print(f'# {len(picks)} free, MemAvailable {avail} GB', file=sys.stderr)
    sys.exit(1)
print(' '.join(picks), avail)
