#!/usr/bin/env bash
# editrun.sh <clone> <out> <suite:target>...: each make target on a clean clone in turn (make -j16), the editpath.sh
# environment; rc per target in <out>/<suite>__<target>.rc, log in .log, a line each in <out>/RUN-<clone>.txt
set -u
C=$1; O=$2; shift 2; mkdir -p "$O"; R=$O/RUN-$(basename $C).txt
export PATH=<tools>/pinned-verilator-5.050:<tools>/sv2v-v0.0.12:<milan>/scratch/a662/make43/bin:/usr/local/bin:/usr/bin:/bin:<tools>/rv32/host/bin
export VERILATOR=<tools>/pinned-verilator-5.050/verilator SV2V=<tools>/sv2v-v0.0.12/sv2v YOSYS=/usr/bin/yosys
export TSN_GEN_ROOT=<management>/tools/tsn-gen-deca300c PYTHONHASHSEED=0
export TMPDIR=<scratch>/tmp/$(basename $C) HOME=<scratch>/wf/home COURSIER_MODE=offline SBT_OPTS=-Dsbt.offline=true
mkdir -p "$TMPDIR"; cd "$C" || exit 3
echo "head $(git rev-parse HEAD) dirty $(git status --porcelain --untracked-files=no | wc -l) start $(date -Iseconds)" | tee -a "$R"
for st in "$@"; do s=${st%%:*}; t=${st#*:}; b=$(date +%s)
  make -C "tb/verilator/$s" -j16 --no-print-directory $t > "$O/${s}__${t}.log" 2>&1; rc=$?
  echo $rc > "$O/${s}__${t}.rc"; echo "$(date +%T) $s $t rc=$rc $(( $(date +%s) - b ))s" | tee -a "$R"
done
echo "porcelain-after $(git status --porcelain --untracked-files=no | wc -l) end $(date -Iseconds)" | tee -a "$R"
