#!/usr/bin/env bash
# unit.sh <name> <width 8|16> <memmax> <log> <cmd...>: run <cmd> as its own user unit milan-927a791-<name>, pinned
# to a CPU range booked through cpu_ranges.py (pick8, or pick --width 16), MemoryMax <memmax>; output to <log>.
# Prints the unit name and the CPUs; returns at once (poll the log and the rc file the command writes).
set -u
N=$1; W=$2; M=$3; L=$4; shift 4
CR=<management>/2026-09-23/cpu_ranges.py
if [ "$W" = 16 ]; then C=$(python3 -I $CR pick --width 16); else C=$(python3 -I $CR pick); fi
[ -n "$C" ] || { echo "no free CPU range"; exit 1; }
systemd-run --user --unit="milan-927a791-$N" --collect -p MemoryMax="$M" -p MemorySwapMax=0 \
  --setenv=HOME=<home> --setenv=PATH=/usr/local/sbin:/usr/local/bin:/usr/bin:/bin \
  /usr/bin/bash -c 'exec /usr/bin/taskset -c "$0" "$@" > "'"$L"'" 2>&1' "$C" "$@"
echo "unit milan-927a791-$N cpus $C log $L"
