#!/usr/bin/env bash
# route_queue4.sh <head>: what the 18:46 crash left of route_queue2.sh, one Vivado build at a time (each through
# vivado-slot.sh): the release image's integrated route from its finished export (rel2/head, export rc 0 in q3/), then
# each split shape with the ROM padded by FILL bytes on the recipe's directive. Every stage's rc in <rt>/q4/<stage>.rc.
set -u
H=$1; RT=<scratch>/rt; T=$RT/tools; Q=$RT/q4; SLOT=<management>/2026-09-23/vivado-slot.sh
FILL=2168
mkdir -p $Q
log() { echo "$(date -Iseconds) $*" | tee -a $Q/queue.log; }
W=$RT/rel2/head
if [ "$(cat $RT/q3/release-export.rc 2>/dev/null)" = 0 ]; then
  log "release route"; $SLOT $T/route_one.sh $W/ax7101/gateware > $Q/release-route.log 2>&1
  cp $W/ax7101/gateware/route.rc $Q/release-route.rc 2>/dev/null
fi
for SH in 8x8 1x1; do
  R=$RT/r2-fill-$SH; W=$RT/f2-$SH
  [ -d $R ] || { $T/clone.sh $R $H > $Q/clone-fill-$SH.log 2>&1; python3 -I $T/fill_patch.py $R $FILL > $Q/fill-patch-$SH.log 2>&1; echo $? > $Q/fill-patch-$SH.rc; }
  log "fill prep $SH"; $T/prep.sh $R $SH $W > $Q/fill-prep-$SH.log 2>&1; cp $W/prep.rc $Q/fill-prep-$SH.rc 2>/dev/null
  if [ "$(cat $W/prep.rc 2>/dev/null)" = 0 ]; then log "fill route $SH"; $SLOT $T/route_one.sh $W/split/gateware > $Q/fill-route-$SH.log 2>&1; cp $W/split/gateware/route.rc $Q/fill-route-$SH.rc 2>/dev/null; fi
done
log "QUEUE DONE"; echo 0 > $Q/queue.rc
