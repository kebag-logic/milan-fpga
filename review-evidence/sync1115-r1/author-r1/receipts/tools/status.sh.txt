#!/bin/bash
# status.sh: one view of this lane's running work.
S=<scratch>
date '+%T'
echo "-- chain"; tail -4 $S/rt/chain.log 2>/dev/null
echo "-- harness"; for f in $S/budget/logs/*.log; do l=$(basename $f .log); echo "$l: started $(grep -c '^COMMAND .*native/Vsim' $f); $(grep -E '^END|groups graded' $f | tr '\n' ' ' | cut -c1-200)"; done
echo "-- workflows"; for b in $S/runs/F/batch-*.log; do echo "$(basename $b): $(grep -c 'rc=' $b) steps; last: $(tail -1 $b | cut -c1-90)"; done
for f in $S/runs/F/logs/*.rc; do [ -f "$f" ] || continue; r=$(cat $f); [ "$r" != 0 ] && echo "FAIL $(basename $f .rc) rc=$r"; done
echo "-- units"; systemctl --user list-units --plain --no-legend 'milan-sync1115-a787-*' | awk '{print $1, $3, $4}'
grep MemAvailable /proc/meminfo; python3 <management>/2026-09-23/cpu_ranges.py show | head -1
