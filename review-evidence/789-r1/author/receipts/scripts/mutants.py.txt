def scrub(text):
    return text.replace(str(WT), "<worktree>").replace(str(SCRATCH), "<scratch>").replace("<home>", "<home>")

def run_code(label, old, new, words):
    root = SCRATCH / label.split()[0]
    shutil.rmtree(root, ignore_errors=True)
    (root / "scripts").mkdir(parents=True)
    (root / ".github").symlink_to(WT / ".github")
    text = GEN
    if old is not None:
        assert text.count(old) == 1, (label, old)
        text = text.replace(old, new)
    (root / "scripts/ci_gate_workflows.py").write_text(text)
    code = ("import sys; sys.path[:0] = [%r, %r]\nimport ci_gate_workflows_selftest as s\ns.fast_tool_controls()\n"
            % (str(root / "scripts"), str(WT / "scripts")))
    done = subprocess.run([sys.executable, "-c", code], cwd=WT, capture_output=True, text=True)
    output = scrub(done.stdout + done.stderr)
    (OUT / (label.split()[0] + ".log")).write_text(output)
    if old is None:
        ok = done.returncode == 0
    else:
        ok = done.returncode != 0 and words in output
    shutil.rmtree(root)
    last = [line for line in output.splitlines() if line.strip()][-1]
    return ok, done.returncode, last

def run_tree(label, kind, words):
    root = SCRATCH / label.split()[0]
    shutil.rmtree(root, ignore_errors=True)
    root.mkdir(parents=True)
    archive = subprocess.run(["git", "-C", str(WT), "archive", "HEAD"], check=True, capture_output=True).stdout
    subprocess.run(["tar", "-x", "-C", str(root)], input=archive, check=True)
    path = root / ".github/workflows/rtl-fast.yml"
    text = path.read_text()
    if kind == "key":
        recipe = text.split("-recipe-", 1)[1].split("-ubuntu-24.04-", 1)[0]
        text = text.replace("-recipe-" + recipe + "-ubuntu-24.04-", "-${{ hashFiles('.github/workflows/*.yml') }}-")
    elif kind == "needs":
        assert text.count("      - yosys-tool\n      - yosys-elaboration\n") == 2
        text = text.replace("      - yosys-tool\n      - yosys-elaboration\n", "      - yosys-elaboration\n", 1)
    elif kind == "miss":
        assert text.count("          fail-on-cache-miss: true\n") == 1
        text = text.replace("          fail-on-cache-miss: true\n", "")
    else:
        text = DEV_WF
    path.write_text(text)
    results = []
    for name, argv, word in (("generator --check", ["scripts/ci_gate_workflows.py", "--check", "--root", str(root)], words[1]),
                             ("ci_events --check", ["scripts/ci_events.py", "--check", "--root", str(root)], words[0])):
        done = subprocess.run([sys.executable, *argv], cwd=WT, capture_output=True, text=True)
        output = scrub(done.stdout + done.stderr)
        (OUT / (label.split()[0] + "-" + name.split()[0] + ".log")).write_text(output)
        results.append((name, done.returncode, done.returncode == 1 and word in output, word))
    shutil.rmtree(root)
    return results

OUT.mkdir(parents=True, exist_ok=True)
rows = ["mutant\trc\tcaught\tfailing words"]
failed = 0
for label, old, new, words in CODE:
    ok, rc, last = run_code(label, old, new, words)
    failed += not ok
    rows.append(f"{label}\t{rc}\t{'yes' if ok else 'NO'}\t{words or 'control: must pass'}")
    print(f"{label}: rc={rc} {'CAUGHT' if old else 'PASS'} " if ok else f"{label}: rc={rc} SURVIVED  {last}")
for label, kind, words in TREE:
    for name, rc, ok, word in run_tree(label, kind, words):
        failed += not ok
        rows.append(f"{label} [{name}]\t{rc}\t{'yes' if ok else 'NO'}\t{word}")
        print(f"{label} [{name}]: rc={rc} {'CAUGHT' if ok else 'SURVIVED'}")
(OUT / "summary.tsv").write_text("\n".join(rows) + "\n")
print("mutants:", "ALL CAUGHT" if not failed else f"{failed} SURVIVED")
sys.exit(1 if failed else 0)
