#!/usr/bin/env python3
"""mut_variant_sizes.py <repo> <out>: each planted mutation of the CPU budget harness that needs a labelled variant,
built with that variant at both 8x8 points (the ledgered size fixture's), its plan read with no budget."""
import sys
from pathlib import Path
repo = Path(sys.argv[1]).resolve(); out = Path(sys.argv[2]).resolve()
for d in ('tb/verilator/ctrl_service_budget', 'sw/litex'):
    sys.path.insert(0, str(repo / d))
import build, ctrl_memory
for n_if in (1, 2):
    for m in sorted(build.MUTATIONS):
        variant = build.MUTATIONS[m][4]
        if variant == 'none':
            continue
        label = f"8x8-if{n_if} mutation={m} variant={variant}"
        try:
            elf = build.firmware(out / f"if{n_if}-{m}", 'endstation_ax7101_8x8', n_if, m, variant)
            plan = ctrl_memory.plan(ctrl_memory.read_elf(Path(elf)), budget=1e9)
            print(f"{label}: rom {plan.rom_bytes} sram {plan.ram_bytes} -> {plan.tiles} RAMB36", flush=True)
        except Exception as e:
            print(f"{label}: REFUSED {type(e).__name__}: {str(e)[:300]}", flush=True)
