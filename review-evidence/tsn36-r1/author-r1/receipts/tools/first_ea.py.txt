#!/usr/bin/env python3
"""first_ea.py RESULTS...: in each CPU budget harness leg's ADP groups, the interface's LINK_UP (the mark of probe
adp.link_up.if0, where the scenario raises the link after a boot with it down) to the first TX commit on the ADP
channel after it (TXC <cycle> <channel> <words>), in ms at the bus clock (100 MHz, the log's CLOCK line). The
harness's ADP channel number is read from its scenario module (scenario.CH['adp'])."""
import re
import sys
from pathlib import Path

sys.path.insert(0, '<lane>/tb/verilator/ctrl_service_budget')
import scenario  # noqa: E402

ADP = scenario.CH['adp']
GROUPS = ('adp-link', 'adp-gm', 'adp-discover', 'adp-advertise', 'control')
for res in map(Path, sys.argv[1:]):
    for g in GROUPS:
        log = res / f'{g}.log'
        if not log.is_file():
            print(f'{res} {g}: no log')
            continue
        text = log.read_text(errors='replace')
        hz = 100_000_000
        m = re.search(r'^CLOCK .*?bus (\d+) MHz', text, re.M)
        if m:
            hz = int(m.group(1)) * 1_000_000
        up = re.search(r'^MARK (\d+) adp\.link_up\.if0$', text, re.M)
        if not up:
            print(f'{res} {g}: no LINK_UP mark')
            continue
        t0 = int(up.group(1))
        tx = [int(c) for c, ch in re.findall(r'^TXC (\d+) (\d+) \d+$', text, re.M) if int(ch) == ADP and int(c) > t0]
        if not tx:
            print(f'{res} {g}: no ADP frame after LINK_UP')
            continue
        print(f'{res.parents[2].name}/{res.parent.parent.name} {g}: LINK_UP at {t0 / hz * 1e3:.3f} ms, first ADP frame '
              f'{(tx[0] - t0) / hz * 1e3:.3f} ms after it')
